      - name: Upload IPA Artifact
        uses: actions/upload-artifact@v4

          name: DriveAssistant-Release-IPA
          path: DriveAssistant-Release.ipa

import AVFoundation
import AudioToolbox
import Foundation
import UIKit

final class AlertManager {
    private var lastWarningAlert = Date.distantPast
    private var lastCriticalAlert = Date.distantPast
    private var lastLaneDepartureAlert = Date.distantPast
    private var lastOverspeedAlert = Date.distantPast
    private let overspeedAlertCooldown: TimeInterval = 10.0
    private let synthesizer = AVSpeechSynthesizer()
    private let audioSession = AVAudioSession.sharedInstance()
    private var interruptionObserver: NSObjectProtocol?
    private let hapticGenerator = UINotificationFeedbackGenerator()

    // Advisory sign alert cooldowns (per key, 8s)
    private var lastAdvisoryAlert: [String: Date] = [:]
    private let advisoryAlertCooldown: TimeInterval = 8.0

    // AVAudioEngine for synthesized chimes
    private lazy var audioEngine = AVAudioEngine()
    private lazy var playerNode = AVAudioPlayerNode()
    private let chimeFrequency: Double = 220.0  // A3 - low pitch
    private let chimeDuration: Double = 0.18
    private var engineStarted = false

    init() {
        configureAudioSession()
        observeInterruptions()
        hapticGenerator.prepare()
        setupChimeEngine()
    }

    deinit {
        if let interruptionObserver {
            NotificationCenter.default.removeObserver(interruptionObserver)
        }
        audioEngine.stop()
        engineStarted = false
    }

    private func configureAudioSession() {
        do {
            try audioSession.setCategory(.playback, mode: .voicePrompt, options: [.duckOthers, .mixWithOthers])
            try audioSession.setActive(true, options: .notifyOthersOnDeactivation)
        } catch {
            print("AlertManager: Failed to configure audio session: \(error)")
        }
    }

    private func ensureEngineRunning() -> Bool {
        if engineStarted && audioEngine.isRunning {
            return true
        }
        do {
            try audioEngine.start()
            engineStarted = true
            return true
        } catch {
            print("AlertManager: Failed to start audio engine: \(error)")
            engineStarted = false
            return false
        }
    }

    private func observeInterruptions() {
        interruptionObserver = NotificationCenter.default.addObserver(
            forName: AVAudioSession.interruptionNotification,
            object: audioSession,
            queue: .main
        ) { [weak self] notification in
            guard let self,
                  let userInfo = notification.userInfo,
                  let typeValue = userInfo[AVAudioSessionInterruptionTypeKey] as? UInt,
                  let type = AVAudioSession.InterruptionType(rawValue: typeValue) else { return }

            if type == .ended {
                if let optionsValue = userInfo[AVAudioSessionInterruptionOptionKey] as? UInt {
                    let options = AVAudioSession.InterruptionOptions(rawValue: optionsValue)
                    if options.contains(.shouldResume) {
                        self.reactivateAudioSession()
                    }
                }
            }
        }
    }

    private func reactivateAudioSession() {
        do {
            try audioSession.setActive(true, options: .notifyOthersOnDeactivation)
        } catch {
            print("AlertManager: Failed to reactivate audio session: \(error)")
        }
    }

    // MARK: - Synthesized Chime Engine
    private func setupChimeEngine() {
        audioEngine.attach(playerNode)
        audioEngine.connect(playerNode, to: audioEngine.mainMixerNode, format: nil)
        _ = ensureEngineRunning()
    }

    private func playLowChime() {
        guard ensureEngineRunning() else { return }
        let sampleRate = audioEngine.mainMixerNode.outputFormat(forBus: 0).sampleRate
        guard sampleRate > 0 else { return }
        let frameCount = AVAudioFrameCount(sampleRate * chimeDuration)
        guard let format = AVAudioFormat(standardFormatWithSampleRate: sampleRate, channels: 1),
              let buffer = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: frameCount) else { return }

        // Validate format matches player node output format
        guard buffer.format.isEqual(playerNode.outputFormat(forBus: 0)) else {
            print("AlertManager: Low chime buffer format mismatch")
            return
        }

        buffer.frameLength = frameCount
        let channelData = buffer.floatChannelData?[0]

        // Generate low-pitched sine wave with quick decay envelope
        for frame in 0..<Int(frameCount) {
            let t = Double(frame) / sampleRate
            let envelope = exp(-t * 8.0)  // Fast decay
            let sample = sin(2.0 * .pi * chimeFrequency * t) * envelope * 0.3
            channelData?[frame] = Float(sample)
        }

        playerNode.scheduleBuffer(buffer, at: nil, options: .interrupts, completionHandler: nil)
        playerNode.play()
    }

    func playWarningAlert() {
        guard Date().timeIntervalSince(lastWarningAlert) >= 5 else { return }
        lastWarningAlert = Date()
        AudioServicesPlaySystemSound(1052)
    }

    func playCriticalAlert(speak: Bool, reduceSpeed: Bool) {
        guard Date().timeIntervalSince(lastCriticalAlert) >= 3 else { return }
        lastCriticalAlert = Date()
        AudioServicesPlaySystemSound(1005)
        hapticGenerator.notificationOccurred(.error)
        guard speak else { return }
        let utterance = AVSpeechUtterance(string: reduceSpeed ? "Giảm tốc độ" : "Giữ khoảng cách an toàn")
        utterance.voice = AVSpeechSynthesisVoice(language: "vi-VN")
        utterance.rate = 0.48
        do {
            try audioSession.setActive(true, options: .notifyOthersOnDeactivation)
            synthesizer.speak(utterance)
        } catch {
            print("AlertManager: Critical speech failed: \(error)")
        }
    }
}

extension AlertManager {
    func playLaneDepartureAlert(speak: Bool) {
        do {
            // Low-pitched double-pulse chime (220Hz A3) - clearly distinct from high-pitched collision beep (~880Hz)
            playLowChime()
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.2) {
                self.playLowChime()
            }

            // Haptic feedback
            hapticGenerator.notificationOccurred(.warning)

            if speak {
                let utterance = AVSpeechUtterance(string: "Cảnh báo lệch làn")
                utterance.voice = AVSpeechSynthesisVoice(language: "vi-VN")
                utterance.rate = 0.5
                try audioSession.setActive(true, options: .notifyOthersOnDeactivation)
                synthesizer.speak(utterance)
            }
        } catch {
            print("AlertManager: Lane departure alert failed: \(error)")
        }
    }

    // MARK: - Stop-and-Go Alert (Lead Vehicle Departure)
    func playStopAndGoAlert(speak: Bool) {
        do {
            // Ascending two-note chime: C5 (523.25 Hz) -> E5 (659.25 Hz), 0.15s apart
            playAscendingChime(note1Freq: 523.25, note2Freq: 659.25, gap: 0.15)

            // Gentle success haptic
            hapticGenerator.notificationOccurred(.success)

            if speak {
                let utterance = AVSpeechUtterance(string: "Xe phía trước đã di chuyển")
                utterance.voice = AVSpeechSynthesisVoice(language: "vi-VN")
                utterance.rate = 0.48
                try audioSession.setActive(true, options: .notifyOthersOnDeactivation)
                synthesizer.speak(utterance)
            }
        } catch {
            print("AlertManager: Stop-and-go alert failed: \(error)")
        }
    }

    private func playAscendingChime(note1Freq: Double, note2Freq: Double, gap: Double) {
        guard ensureEngineRunning() else { return }
        let sampleRate = audioEngine.mainMixerNode.outputFormat(forBus: 0).sampleRate
        guard sampleRate > 0 else { return }
        let noteDuration = 0.12
        let frameCount = AVAudioFrameCount(sampleRate * noteDuration)

        guard let format = AVAudioFormat(standardFormatWithSampleRate: sampleRate, channels: 1) else { return }

        // Validate format matches player node output format
        guard format.isEqual(playerNode.outputFormat(forBus: 0)) else {
            print("AlertManager: Ascending chime format mismatch")
            return
        }

        // Note 1
        if let buffer1 = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: frameCount) {
            buffer1.frameLength = frameCount
            let channelData = buffer1.floatChannelData?[0]
            for frame in 0..<Int(frameCount) {
                let t = Double(frame) / sampleRate
                let envelope = exp(-t * 10.0)
                let sample = sin(2.0 * .pi * note1Freq * t) * envelope * 0.25
                channelData?[frame] = Float(sample)
            }
            playerNode.scheduleBuffer(buffer1, at: nil, options: .interrupts, completionHandler: nil)
        }

        // Note 2 (delayed)
        DispatchQueue.main.asyncAfter(deadline: .now() + gap) { [weak self] in
            guard let self,
                  self.ensureEngineRunning(),
                  let buffer2 = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: frameCount) else { return }
            buffer2.frameLength = frameCount
            let channelData = buffer2.floatChannelData?[0]
            for frame in 0..<Int(frameCount) {
                let t = Double(frame) / sampleRate
                let envelope = exp(-t * 10.0)
                let sample = sin(2.0 * .pi * note2Freq * t) * envelope * 0.25

    private func computeAverageLuma(pixelBuffer: CVPixelBuffer) -> Double {
        CVPixelBufferLockBaseAddress(pixelBuffer, .readOnly)
        defer { CVPixelBufferUnlockBaseAddress(pixelBuffer, .readOnly) }
        
        guard let baseAddress = CVPixelBufferGetBaseAddress(pixelBuffer) else { return 1.0 }
        let width = CVPixelBufferGetWidth(pixelBuffer)
        let height = CVPixelBufferGetHeight(pixelBuffer)
        let bytesPerRow = CVPixelBufferGetBytesPerRow(pixelBuffer)
        
        // Sample every 8th pixel for performance
        var totalLuma: Double = 0
        var sampleCount = 0
        let sampleStride = 8
        
        for y in stride(from: 0, to: height, by: sampleStride) {
            let rowPtr = baseAddress.advanced(by: y * bytesPerRow).assumingMemoryBound(to: UInt8.self)
            for x in stride(from: 0, to: width, by: sampleStride) {
                let pixelOffset = x * 4 // Assuming BGRA/ARGB 32-bit
                if pixelOffset + 2 < bytesPerRow {
                    let b = Double(rowPtr[pixelOffset])
                    let g = Double(rowPtr[pixelOffset + 1])
                    let r = Double(rowPtr[pixelOffset + 2])
                    // Standard luma coefficients (Rec. 709)
                    let luma = 0.2126 * r + 0.7152 * g + 0.0722 * b
                    totalLuma += luma
                    sampleCount += 1
                }
            }
        }
        
        guard sampleCount > 0 else { return 1.0 }
        let avgLuma = totalLuma / Double(sampleCount)
        // Normalize to EV-like scale (0-255 -> roughly -3 to +3 EV)
        return (avgLuma / 255.0) * 6.0 - 3.0
    }

        diagnostics?.record(type: "slow_inference", details: "durationSeconds=\(duration), thermalState=\(String(describing: thermalStateForInference()))")
    }

    private func recordWarmupDuration(_ duration: TimeInterval) {
        guard !warmupComplete else { return }
        warmupFrameCount += 1
        warmupTotalDuration += duration
        guard warmupFrameCount >= 5 else { return }
        let averageMilliseconds = warmupTotalDuration / Double(warmupFrameCount) * 1000
        baselineInferenceFPS = averageMilliseconds < 40 ? 10 : 6
        warmupComplete = true
        diagnostics?.record(type: "vision_warmup_completed", details: "frames=5, averageLatencyMS=\(averageMilliseconds), baselineFPS=\(baselineInferenceFPS), computeUnits=cpuAndNeuralEngine")
        logger.info("Vision warm-up complete: average latency \(averageMilliseconds, privacy: .public) ms, baseline \(self.baselineInferenceFPS, privacy: .public) FPS")
    }

        switch state {
        case .critical:
            return nil
        case .serious:
            return seriousInferenceInterval
        case .fair:
            return fairInferenceInterval
        case .nominal:
            return 1.0 / baselineInferenceFPS
        @unknown default:
            return 1.0 / baselineInferenceFPS
        }
    }

    private func recordPerformanceIfNeeded() {

    private func handle(request: VNRequest, pixelBuffer: CVPixelBuffer) {
        let now = CACurrentMediaTime()
        let candidates = (request.results as? [VNRecognizedObjectObservation] ?? []).compactMap { observation -> (CGRect, Float, String)? in
            guard let label = observation.labels.first,
                  ["car", "truck", "bus", "motorcycle", "motorbike"].contains(label.identifier) else { return nil }
            guard label.confidence >= 0.42 else { return nil }
            let box = observation.boundingBox
            guard box.width >= 0.032, box.height >= 0.032 else { return nil }
            return (box, label.confidence, label.identifier)
        }
        
        let hasLockedTarget = trackedBox != nil
        let laneOffset = currentLaneCenterOffsetX
        let horizonEnabled = isHorizonFilterEnabled
        let horizonY = currentHorizonY
        
        let effectiveHorizonForFilter = horizonEnabled ? max(horizonY, 0.48) : horizonY
        let ceilingY = min(effectiveHorizonForFilter, 0.45) - 0.03
        
        let horizonFiltered = mergeOverlappingDetections(candidates).filter { observation in
            let box = observation.0
            
            if horizonEnabled {
                let boxBottomYFromTop = 1.0 - box.minY
                if boxBottomYFromTop < ceilingY {
                    return false
                }
            }
            return true
        }
        
        var inROI: [(CGRect, Float, String)] = []
        var outOfROI: [(CGRect, Float, String)] = []
        
        // Ego-vehicle hood exclusion: discard wide detections touching bottom of frame
        let hoodFiltered = horizonFiltered.filter { observation in
            let box = observation.0
            let boxBottomYFromTop = 1.0 - box.minY
            return !(boxBottomYFromTop >= 0.88 && box.width > 0.55)
        }
        
        // Get dynamic lane edges from LaneDetector (normalized 0...1, top-left origin)
        // Note: laneDetector processes frames at a lower rate, so check if it has recent data
        let dynamicLaneEdgesAvailable = laneDetector?.leftLaneConfidence ?? 0 > 0.5 && laneDetector?.rightLaneConfidence ?? 0 > 0.5

        for observation in hoodFiltered {
            let box = observation.0

            let bottomYFromTop = 1.0 - box.minY
            var isInLane = false

            let laneCenterX = 0.5 + laneOffset
            let isLeftOfLaneCenter = box.midX < laneCenterX

            if dynamicLaneEdgesAvailable,
               let laneEdges = laneDetector?.getLaneEdgesAtY(bottomYFromTop, imageSize: cameraFrameSize) {
                // Dynamic lane edges from detected road markings
                let leftEdge = laneEdges.left
                let rightEdge = laneEdges.right

                // Check if box center is within lane edges OR 40% bottom edge overlap
                let centerInside = box.midX >= leftEdge && box.midX <= rightEdge
                let boxLeft = box.minX
                let boxRight = box.maxX
                let boxWidth = boxRight - boxLeft
                if boxWidth > 0 {
                    let overlapLeft = max(boxLeft, leftEdge)
                    let overlapRight = min(boxRight, rightEdge)
                    let overlapWidth = max(0, overlapRight - overlapLeft)
                    let edgeOverlap = overlapWidth / boxWidth >= 0.40
                    // Strict left-edge: if vehicle is left of lane center, require centerInside (no edgeOverlap)
                    if isLeftOfLaneCenter {
                        isInLane = centerInside
                    } else {
                        isInLane = centerInside || edgeOverlap
                    }
                }
            } else {
                // Fallback to static trapezoid ROI
                let bottomCenterInside = Self.isInsideForwardLaneROI(x: box.midX, yFromTop: bottomYFromTop,
                                                                      laneCenterOffsetX: laneOffset,
                                                                      truncateTopY: nil)
                let bottomEdgeOverlap = Self.hasSufficientBottomEdgeOverlap(box: box, laneCenterOffsetX: laneOffset, bottomYFromTop: bottomYFromTop)
                // Strict left-edge: if vehicle is left of lane center, require bottomCenterInside (no edgeOverlap)
                if isLeftOfLaneCenter {
                    isInLane = bottomCenterInside
                } else {
                    isInLane = bottomCenterInside || bottomEdgeOverlap
                }
            }

            if isInLane {
                inROI.append(observation)
            } else {
                outOfROI.append(observation)
            }
        }
        
        DispatchQueue.main.async {
            self.inROIVehicles = inROI
            self.outOfROIVehicles = outOfROI
        }
        
        let selected = selectLeadVehicle(from: inROI)
        logDetectionFrame(inROI, selected: selected)
        guard let selected else {
            if trackedBox != nil && now - trackedBoxTimestamp < 0.20 {
                return
            }
            self.trackedBox = nil
            DispatchQueue.main.async {
                self.leadVehicleDistance = nil
                self.leadVehicleBox = nil
                self.leadVehicleConfidence = 0
            }
            return
        }
        let smoothedBox = smoothTrackedBox(selected.0)
        let realWidth: Double
        switch selected.2 {
        case "car": realWidth = 1.8
        case "truck", "bus": realWidth = 2.5
        case "motorcycle", "motorbike": realWidth = 0.75
        default: realWidth = 1.8
        }
        calibrationLock.lock()
        let factor = calibrationFactor
        calibrationLock.unlock()
        let normalizedBoxWidth = Double(smoothedBox.width)
        guard normalizedBoxWidth > 0 else { return }
        let distance = (realWidth * factor) / normalizedBoxWidth
        let selectedConfidence = selected.1
        DispatchQueue.main.async {
            self.leadVehicleDistance = distance
            self.leadVehicleBox = smoothedBox
            self.leadVehicleConfidence = selectedConfidence
        }
    }
    private func selectLeadVehicle(from observations: [(CGRect, Float, String)]) -> (CGRect, Float, String)? {
        guard !observations.isEmpty else {
            pendingBox = nil
            pendingBoxFrames = 0
            return nil
        }
        if let trackedBox {
            let matching = observations.filter { isPlausibleContinuation(from: trackedBox, to: $0.0) }
            if let closestMatch = closestVehicle(in: matching) {
                pendingBox = nil
                pendingBoxFrames = 0
                return closestMatch
            }

            guard let candidate = closestVehicle(in: observations) else { return nil }
            if let pendingBox, isPlausiblePendingMatch(from: pendingBox, to: candidate.0) {
                self.pendingBox = candidate.0
                pendingBoxFrames += 1
            } else {
                pendingBox = candidate.0
                pendingBoxFrames = 1
            }
            guard pendingBoxFrames >= 3 else { return nil }
            self.trackedBox = nil
            pendingBox = nil
            pendingBoxFrames = 0
            diagnostics?.record(type: "target_switched", details: "newBox=\(candidate.0.debugDescription), confidence=\(candidate.1), label=\(candidate.2)")
            // Oncoming lane suspect diagnostic
            let laneOffset = currentLaneCenterOffsetX
            if candidate.0.midX < (0.5 + laneOffset) - 0.035 {
                diagnostics?.record(type: "oncoming_lane_suspect", details: "midX=\(String(format: "%.3f", candidate.0.midX)), box=\(candidate.0.debugDescription), laneOffset=\(String(format: "%.3f", laneOffset)), confidence=\(candidate.1), label=\(candidate.2)")
            }
            return candidate
        }
        pendingBox = nil
        pendingBoxFrames = 0
        let candidate = closestVehicle(in: observations)
        // Oncoming lane suspect diagnostic for first selection
        if let candidate {
            let laneOffset = currentLaneCenterOffsetX
            if candidate.0.midX < (0.5 + laneOffset) - 0.035 {
                diagnostics?.record(type: "oncoming_lane_suspect", details: "midX=\(String(format: "%.3f", candidate.0.midX)), box=\(candidate.0.debugDescription), laneOffset=\(String(format: "%.3f", laneOffset)), confidence=\(candidate.1), label=\(candidate.2)")
            }
        }
        return candidate
    }

    private func isPlausibleContinuation(from previous: CGRect, to candidate: CGRect) -> Bool {
        if intersectionOverUnion(previous, candidate) >= 0.10 {
            return true
        }
        let widthRatio = candidate.width / max(previous.width, 0.001)
        let heightRatio = candidate.height / max(previous.height, 0.001)
        return centerDistance(previous, candidate) <= 0.10 &&
            abs(previous.midY - candidate.midY) <= 0.14 &&
            (0.55...1.80).contains(widthRatio) &&
            (0.55...1.80).contains(heightRatio)
    }

    private func isPlausiblePendingMatch(from previous: CGRect, to candidate: CGRect) -> Bool {
        intersectionOverUnion(previous, candidate) >= 0.05 || centerDistance(previous, candidate) <= 0.14
    }

    private func logDetectionFrame(_ observations: [(CGRect, Float, String)], selected: (CGRect, Float, String)?) {
        let now = CACurrentMediaTime()
        guard now - lastDetectionLogTime >= 1 else { return }
        lastDetectionLogTime = now
        let candidates = observations.map {
            String(format: "%@ conf=%.2f box=(%.3f,%.3f,%.3f,%.3f)", $0.2, $0.1, $0.0.minX, $0.0.minY, $0.0.width, $0.0.height)
        }.joined(separator: ";")
        let selectedDetails = selected.map {
            String(format: "%@ conf=%.2f box=(%.3f,%.3f,%.3f,%.3f)", $0.2, $0.1, $0.0.minX, $0.0.minY, $0.0.width, $0.0.height)
        } ?? "none"
        diagnostics?.record(type: "detection_frame", details: "candidateCount=\(observations.count), candidates=[\(candidates)], selected=\(selectedDetails), tracked=\(trackedBox?.debugDescription ?? "none"), pendingFrames=\(pendingBoxFrames)")
    }

    private func closestVehicle(in observations: [(CGRect, Float, String)]) -> (CGRect, Float, String)? {
        observations.max { lhs, rhs in
            let lhsBottom = 1 - lhs.0.minY
            let rhsBottom = 1 - rhs.0.minY
            if abs(lhsBottom - rhsBottom) < 0.01 {
                return lhs.0.width < rhs.0.width
            }
            return lhsBottom < rhsBottom
        }
    }

    private func mergeOverlappingDetections(_ observations: [(CGRect, Float, String)]) -> [(CGRect, Float, String)] {
        var merged: [(CGRect, Float, String)] = []
        for observation in observations {
            guard let index = merged.firstIndex(where: { intersectionOverUnion($0.0, observation.0) > 0.30 }) else {
                merged.append(observation)
                continue
            }
            let existing = merged[index]
            let label = existing.1 >= observation.1 ? existing.2 : observation.2
            merged[index] = (existing.0.union(observation.0), max(existing.1, observation.1), label)
        }
        return merged
    }

    private func smoothTrackedBox(_ box: CGRect) -> CGRect {
        let alpha: CGFloat = 0.35
        let smoothed: CGRect
        if let trackedBox {
            smoothed = CGRect(
                x: trackedBox.minX + (box.minX - trackedBox.minX) * alpha,
                y: trackedBox.minY + (box.minY - trackedBox.minY) * alpha,
                width: trackedBox.width + (box.width - trackedBox.width) * alpha,
                height: trackedBox.height + (box.height - trackedBox.height) * alpha
            )
        } else {
            smoothed = box
        }
        trackedBox = smoothed
        trackedBoxTimestamp = CACurrentMediaTime()
        return smoothed
    }

    private func intersectionOverUnion(_ lhs: CGRect, _ rhs: CGRect) -> CGFloat {
        let intersection = lhs.intersection(rhs)
        guard !intersection.isNull, intersection.width > 0, intersection.height > 0 else { return 0 }
        let union = lhs.area + rhs.area - intersection.area
        return union > 0 ? intersection.area / union : 0
    }

    private func centerDistance(_ lhs: CGRect, _ rhs: CGRect) -> CGFloat {
        hypot(lhs.midX - rhs.midX, lhs.midY - rhs.midY)
    }

    // MARK: - Forward Lane ROI
    
    private static func isInsideForwardLaneROI(x: CGFloat, yFromTop: CGFloat,
                                                laneCenterOffsetX: Double,
                                                truncateTopY: CGFloat? = nil) -> Bool {
        let topY: CGFloat = 0.38
        let bottomY: CGFloat = 0.95
        let topWidth: CGFloat = 0.16   // narrowed from 0.22 for narrow two-way streets
        let bottomWidth: CGFloat = 0.42 // narrowed from 0.48
        
        guard yFromTop >= topY && yFromTop <= bottomY else { return false }
        
        let progress = (yFromTop - topY) / (bottomY - topY)
        let centerX = 0.5 + laneCenterOffsetX
        let halfTopWidth = topWidth / 2
        let halfBottomWidth = bottomWidth / 2
        
        let left = centerX - (halfTopWidth + (halfBottomWidth - halfTopWidth) * progress)
        let right = centerX + (halfTopWidth + (halfBottomWidth - halfTopWidth) * progress)
        
        if let truncateY = truncateTopY, yFromTop < truncateY {
            return false
        }
        
        return x >= left && x <= right
    }
    
    private static func hasSufficientBottomEdgeOverlap(box: CGRect, laneCenterOffsetX: Double, bottomYFromTop: CGFloat) -> Bool {
        let topY: CGFloat = 0.38
        let bottomY: CGFloat = 0.95
        let topWidth: CGFloat = 0.16   // narrowed from 0.22
        let bottomWidth: CGFloat = 0.42 // narrowed from 0.48
        
        let boxBottomY = box.minY
        let yFromTop = bottomYFromTop
        guard yFromTop >= topY && yFromTop <= bottomY else { return false }
        
        let progress = (yFromTop - topY) / (bottomY - topY)
        let centerX = 0.5 + laneCenterOffsetX
        let halfTopWidth = topWidth / 2
        let halfBottomWidth = bottomWidth / 2
        
        let trapLeft = centerX - (halfTopWidth + (halfBottomWidth - halfTopWidth) * progress)
        let trapRight = centerX + (halfTopWidth + (halfBottomWidth - halfTopWidth) * progress)
        
        let boxLeft = box.minX
        let boxRight = box.maxX
        let boxWidth = boxRight - boxLeft
        guard boxWidth > 0 else { return false }
        
        let overlapLeft = max(boxLeft, trapLeft)
        let overlapRight = min(boxRight, trapRight)
        let overlapWidth = max(0, overlapRight - overlapLeft)
        
        return overlapWidth / boxWidth >= 0.40
    }
}

    var area: CGFloat { width * height }
}

                .compactMap({ ($0 as? UIWindowScene)?.keyWindow }).first else {
                isCapturingScreenshot = false
                return
            }

            let bounds = window.bounds
            let renderer = UIGraphicsImageRenderer(bounds: bounds)

            // First get the camera frame
            let cameraImage = model.camera.snapshotCameraImage()

            let composited = renderer.image { ctx in
                // 1. Draw camera image (aspectFill to match CameraPreview)
                if let camImg = cameraImage {
                    let videoSize = model.camera.cameraFrameSize
                    let scale = max(bounds.width / videoSize.width, bounds.height / videoSize.height)
                    let scaledVideoSize = CGSize(width: videoSize.width * scale, height: videoSize.height * scale)
                    let xOffset = (bounds.width - scaledVideoSize.width) / 2
                    let yOffset = (bounds.height - scaledVideoSize.height) / 2
                    camImg.draw(in: CGRect(x: xOffset, y: yOffset, width: scaledVideoSize.width, height: scaledVideoSize.height))
                }

                // 2. Draw the HUD overlay (window hierarchy) - skip CameraPreview black background
                window.drawHierarchy(in: bounds, afterScreenUpdates: true)
            }

            isCapturingScreenshot = false

            // Save to photo library
            PHPhotoLibrary.requestAuthorization(for: .addOnly) { status in
                guard status == .authorized else { return }
                PHPhotoLibrary.shared().performChanges({
                    PHAssetChangeRequest.creationRequestForAsset(from: composited)
                }) { success, _ in
                    if success && !silent {
                        DispatchQueue.main.async { screenshotSaved = true }
                    }
                }
            }
        }
    }

    var body: some View {
        GeometryReader { proxy in
            let screenBox = model.camera.leadVehicleBox.map { screenRect(forVisionBox: $0, in: proxy.size) }
            ZStack {
                if model.ecoBlackHUD || model.hudMirrorMode {
                    Color.black
                } else if isCapturingScreenshot {
                    Color.clear
                } else {
                    CameraPreview(camera: model.camera)
                }

                if model.showHorizonGuide && model.motion.isActive {
                    HorizonOverlay(motion: model.motion)
                }

                LaneCarpet(cameraFrameSize: model.camera.cameraFrameSize, laneDetector: model.laneDetector, vehicleRect: screenBox, horizonY: model.motion.effectiveHorizonY, laneCenterOffsetX: model.laneCenterOffsetX, isLDWSEnabled: model.enableLaneDetection)

                if let screenBox {
                    VehicleTargetOverlay(screenRect: screenBox, distance: model.distance, color: activeStateColor)
                }

                // Out-of-ROI vehicles: subtle gray boxes (no alerts)
                if model.showOutOfROIVehicles {
                    ForEach(Array(model.camera.outOfROIVehicles.enumerated()), id: \.offset) { _, item in
                        let (box, conf, label) = item
                        let screenRect = screenRect(forVisionBox: box, in: proxy.size)
                        Rectangle()
                            .stroke(Color.gray.opacity(0.35), lineWidth: 1)
                            .frame(width: screenRect.width, height: screenRect.height)
                            .position(x: screenRect.midX, y: screenRect.midY)
                            .allowsHitTesting(false)
                    }
                }

                hud
            }
        }
        .background(Color.black)
        .ignoresSafeArea(.all)
        .scaleEffect(x: 1, y: model.hudMirrorMode ? -1 : 1)
        .sheet(isPresented: $model.isSettingsPresented) { SettingsView(model: model) }
        .statusBarHidden(true)
        .onAppear { model.start() }
        .onDisappear { model.stop() }
        .overlay(alignment: .top) {
            if model.camera.cameraAuthorizationStatus == .denied || model.camera.cameraAuthorizationStatus == .restricted {
                CameraPermissionBanner()
            }
        }
        .overlay(alignment: .topTrailing) {
            if screenshotFlash {
                Color.white.opacity(0.3)
                    .ignoresSafeArea()
                    .transition(.opacity)
            }
        }

            Button("OK", role: .cancel) {}
        }
    }

    private func screenRect(forVisionBox box: CGRect, in screenSize: CGSize) -> CGRect {
        let videoSize = model.camera.cameraFrameSize
        let scale = max(screenSize.width / videoSize.width, screenSize.height / videoSize.height)
        let scaledVideoSize = CGSize(width: videoSize.width * scale, height: videoSize.height * scale)
        let xOffset = (screenSize.width - scaledVideoSize.width) / 2
        let yOffset = (screenSize.height - scaledVideoSize.height) / 2
        let rawRect = CGRect(
            x: box.minX * videoSize.width,
            y: (1 - box.maxY) * videoSize.height,
            width: box.width * videoSize.width,
            height: box.height * videoSize.height
        )
        return CGRect(
            x: rawRect.minX * scale + xOffset,
            y: rawRect.minY * scale + yOffset,
            width: rawRect.width * scale,
            height: rawRect.height * scale
        )
    }

private struct DistanceCard: View {
    @ObservedObject var model: DriveAssistantModel
    let color: Color
    let nightDim: Bool

    private var progress: Double {
        guard let distance = model.distance else { return 0 }
        return min(max(distance / max(model.activeMinimumDistance * 1.5, 1), 0), 1)
    }

    private var statusText: String {
        if model.usesVietnamese {
            switch model.alertState {
            case .normal: return "Đủ khoảng cách"
            case .warning: return "Thiếu khoảng cách"
            case .critical: return "Quá gần"
            }
        } else {
            switch model.alertState {
            case .normal: return "Safe gap"
            case .warning: return "Gap closing"
            case .critical: return "Too close"
            }
        }
    }

    // Calibration status text for HUD
    private var calibrationStatusText: String? {
        let motion = model.motion
        if motion.isMountShifted {
            return model.usesVietnamese ? "⚠ Máy bị xê dịch" : "⚠ Mount shifted"
        } else if !motion.isAutoCalibrated && model.speedKMPH > 25 {
            return model.usesVietnamese ? "Đang tự hiệu chỉnh \(motion.calibrationProgress)%" : "Auto-calibrating \(motion.calibrationProgress)%"
        }
        return nil
    }

    private var calibrationStatusColor: Color {
        if model.motion.isMountShifted {
            return .orange
        }
        return .cyan
    }

    // Stop-and-Go alert status (emerald when active)
    private var stopAndGoStatusText: String? {
        guard model.isLeadVehicleMovingAlertActive else { return nil }
        return model.usesVietnamese ? "Xe trước đã di chuyển" : "Lead vehicle moving"
    }

    private var stopAndGoStatusColor: Color {
        Color(red: 0.0, green: 0.902, blue: 0.463)  // Emerald
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            HStack(alignment: .lastTextBaseline, spacing: 7) {
                Text(model.distance.map { String(format: "%.0f", $0) } ?? "--")
                    .font(.system(size: 54, weight: .bold, design: .rounded))
                    .monospacedDigit()
                    .foregroundColor(.white)
                Text("m")
                    .font(.system(size: 22, weight: .bold, design: .rounded))
                    .foregroundColor(.white.opacity(0.9))
            }
            Text(statusText)
                .font(.system(size: 13, weight: .bold, design: .rounded))
                .foregroundColor(color)
            SegmentedProgressBar(progress: progress, color: color, activeMinimumDistance: model.activeMinimumDistance)

            // Calibration status indicator (only when relevant)
            if let statusText = calibrationStatusText {
                Text(statusText)
                    .font(.system(size: 11, weight: .medium, design: .rounded))
                    .foregroundColor(calibrationStatusColor)
                    .transition(.opacity.combined(with: .move(edge: .top)))
            }

            // Stop-and-Go alert status
            if let statusText = stopAndGoStatusText {
                Text(statusText)
                    .font(.system(size: 11, weight: .medium, design: .rounded))
                    .foregroundColor(stopAndGoStatusColor)
                    .transition(.opacity.combined(with: .move(edge: .top)))
            }
        }
        .padding(18)
        .frame(width: 245, alignment: .leading)
        .glassCard(cornerRadius: 20, enabled: model.glassHUD, nightDim: nightDim)
        .animation(.easeInOut(duration: 0.3), value: calibrationStatusText)
        .animation(.easeInOut(duration: 0.3), value: stopAndGoStatusText)
    }
}

private struct SegmentedProgressBar: View {
    let progress: Double
    let color: Color
    let activeMinimumDistance: Double

    private var thresholdRatio: Double {
        activeMinimumDistance / (activeMinimumDistance * 1.5)
    }

    var body: some View {
        GeometryReader { proxy in
            let spacing: CGFloat = 4
            let segmentWidth = (proxy.size.width - spacing * 9) / 10
            ZStack(alignment: .leading) {
                HStack(spacing: spacing) {
                    ForEach(0..<10, id: \.self) { index in
                        RoundedRectangle(cornerRadius: 2)
                            .fill(Double(index) / 10 < progress ? color : Color.white.opacity(0.20))
                            .frame(width: segmentWidth, height: 6)
                    }
                }
                Rectangle()
                    .fill(Color.white)
                    .frame(width: 2, height: 14)
                    .offset(x: proxy.size.width * thresholdRatio - 1)
            }
        }
        .frame(height: 14)
    }
}

private struct LaneCarpet: View {
    let cameraFrameSize: CGSize
    @ObservedObject var laneDetector: LaneDetector
    let vehicleRect: CGRect?
    let horizonY: Double
    let laneCenterOffsetX: Double
    let isLDWSEnabled: Bool

    var body: some View {
        GeometryReader { proxy in
            let canvasSize = proxy.size
            
            // Use dynamic lanes if detected with good confidence, else static fallback
            let (leftPath, rightPath, fillPath, leftColor, rightColor, showLeftAlert, showRightAlert) = buildLanePaths(canvasSize: canvasSize, cameraFrameSize: cameraFrameSize)
            
            ZStack {
                // Interior fill
                fillPath
                    .fill(
                        LinearGradient(
                            colors: [.cyan.opacity(0.08), .cyan.opacity(0.0)],
                            startPoint: .bottom,
                            endPoint: .top
                        )
                    )
                
                // Left lane line - color based on state: cyan (normal) -> amber (drifting) -> red (alert)
                leftPath
                    .stroke(
                        LinearGradient(
                            colors: [leftColor.opacity(0.55), leftColor.opacity(0.0)],
                            startPoint: .bottom,
                            endPoint: .top
                        ),
                        style: StrokeStyle(lineWidth: 2.5, lineCap: .round)
                    )
                
                // Right lane line
                rightPath
                    .stroke(
                        LinearGradient(
                            colors: [rightColor.opacity(0.55), rightColor.opacity(0.0)],
                            startPoint: .bottom,
                            endPoint: .top
                        ),
                        style: StrokeStyle(lineWidth: 2.5, lineCap: .round)
                    )
                
                // Departure pulsing alert on crossed rail
                if showLeftAlert {
                    leftPath
                        .stroke(Color.red.opacity(0.9), style: StrokeStyle(lineWidth: 5, lineCap: .round, dash: [6, 3]))
                        .animation(.easeInOut(duration: 0.4).repeatForever(autoreverses: true), value: laneDetector.departureAlertActive)
                }
                if showRightAlert {
                    rightPath
                        .stroke(Color.red.opacity(0.9), style: StrokeStyle(lineWidth: 5, lineCap: .round, dash: [6, 3]))
                        .animation(.easeInOut(duration: 0.4).repeatForever(autoreverses: true), value: laneDetector.departureAlertActive)
                }
            }
        }
        .ignoresSafeArea(.all)
        .allowsHitTesting(false)
    }
    
        private func buildLanePaths(canvasSize: CGSize, cameraFrameSize: CGSize) -> (left: Path, right: Path, fill: Path, leftColor: Color, rightColor: Color, showLeftAlert: Bool, showRightAlert: Bool) {
        var leftPath = Path()
        var rightPath = Path()
        var fillPath = Path()
        
        let hasDynamicLanes = laneDetector.isLaneDetectionActive &&
                              laneDetector.leftLaneConfidence > 0.5 &&
                              laneDetector.rightLaneConfidence > 0.5 &&
                              !laneDetector.leftLanePoints.isEmpty &&
                              !laneDetector.rightLanePoints.isEmpty
        
        // Determine colors based on departure state
        let leftColor: Color
        let rightColor: Color
        let showLeftAlert: Bool
        let showRightAlert: Bool
        
        switch laneDetector.departureState {
        case .driftingLeft:
            leftColor = .orange
            rightColor = .cyan
            showLeftAlert = laneDetector.departureAlertActive
            showRightAlert = false
        case .driftingRight:
            leftColor = .cyan
            rightColor = .orange
            showLeftAlert = false
            showRightAlert = laneDetector.departureAlertActive
        case .none:
            leftColor = .cyan
            rightColor = .cyan
            showLeftAlert = false
            showRightAlert = false
        }
        
        // Clip lane carpet at lead vehicle's bottom edge (rear bumper)
        let horizonTopY = canvasSize.height * horizonY
        let bottomY = canvasSize.height
        let effectiveTopY = vehicleRect.map { max(horizonTopY, min(bottomY, $0.maxY)) } ?? horizonTopY
        
        if hasDynamicLanes {
            // Aspect-fill transform matching screenRect(forVisionBox:in:)
            let videoSize = cameraFrameSize
            let scale = max(canvasSize.width / videoSize.width, canvasSize.height / videoSize.height)
            let scaledVideoSize = CGSize(width: videoSize.width * scale, height: videoSize.height * scale)
            let xOffset = (canvasSize.width - scaledVideoSize.width) / 2
            let yOffset = (canvasSize.height - scaledVideoSize.height) / 2
            func transformPoint(_ pt: CGPoint) -> CGPoint {
                // pt is normalized (0..1) top-left origin from LaneDetector
                // Convert directly to raw video pixel coordinates (no Y-flip needed)
                let rawX = pt.x * videoSize.width
                let rawY = pt.y * videoSize.height
                // Apply aspect-fill scale and offset
                return CGPoint(x: rawX * scale + xOffset, y: rawY * scale + yOffset)
            }

            var leftPoints = laneDetector.leftLanePoints.map(transformPoint)
            var rightPoints = laneDetector.rightLanePoints.map(transformPoint)

            // Clip dynamic curved rails to effectiveTopY (interpolate at boundary)
            leftPoints = clipPointsToY(leftPoints, minY: effectiveTopY)
            rightPoints = clipPointsToY(rightPoints, minY: effectiveTopY)

            if !leftPoints.isEmpty && !rightPoints.isEmpty {
                leftPath.move(to: leftPoints[0])
                for i in 1..<leftPoints.count {
                    leftPath.addLine(to: leftPoints[i])
                }

                rightPath.move(to: rightPoints[0])
                for i in 1..<rightPoints.count {
                    rightPath.addLine(to: rightPoints[i])
                }

                fillPath.move(to: leftPoints[0])
                for point in leftPoints { fillPath.addLine(to: point) }
                for point in rightPoints.reversed() { fillPath.addLine(to: point) }
                fillPath.closeSubpath()

                return (leftPath, rightPath, fillPath, leftColor, rightColor, showLeftAlert, showRightAlert)
            }
            // Fallback to static if clipping leaves nothing
        }

        // Static fallback trapezoid in NORMALIZED VIDEO SPACE (0..1, top-left origin)
        // Matches CameraManager.isInsideForwardLaneROI static ROI and model.laneCenterOffsetX
        let videoSize = cameraFrameSize
        let scale = max(canvasSize.width / videoSize.width, canvasSize.height / videoSize.height)
        let scaledVideoSize = CGSize(width: videoSize.width * scale, height: videoSize.height * scale)
        let xOffset = (canvasSize.width - scaledVideoSize.width) / 2
        let yOffset = (canvasSize.height - scaledVideoSize.height) / 2

        func projectNorm(_ norm: CGPoint) -> CGPoint {
            let rawX = norm.x * videoSize.width
            let rawY = norm.y * videoSize.height
            return CGPoint(x: rawX * scale + xOffset, y: rawY * scale + yOffset)
        }

        let centerXNorm = 0.5 + laneCenterOffsetX
        let horizonTopYNorm = max(0.25, min(0.65, horizonY))
        let bottomYNorm = 0.95
        let topWidthNorm = 0.16
        let bottomWidthNorm = 0.42

        // Compute effectiveTopY in normalized video space (clip at lead vehicle bottom)
        let effectiveTopYNorm: CGFloat
        if let vRect = vehicleRect {
            // Convert screen vehicleRect.maxY back to normalized video Y
            let rawY = (vRect.maxY - yOffset) / scale
            let normY = rawY / videoSize.height
            effectiveTopYNorm = max(horizonTopYNorm, min(bottomYNorm, normY))
        } else {
            effectiveTopYNorm = horizonTopYNorm
        }

        // Interpolate top width at effectiveTopYNorm to preserve true perspective angle
        let totalHeightNorm = bottomYNorm - horizonTopYNorm
        let tNorm = totalHeightNorm > 0 ? (effectiveTopYNorm - horizonTopYNorm) / totalHeightNorm : 0
        let clippedTopWidthNorm = topWidthNorm + (bottomWidthNorm - topWidthNorm) * tNorm

        // Four corners in normalized video space
        let leftTopNorm = CGPoint(x: centerXNorm - clippedTopWidthNorm / 2, y: effectiveTopYNorm)
        let rightTopNorm = CGPoint(x: centerXNorm + clippedTopWidthNorm / 2, y: effectiveTopYNorm)
        let leftBottomNorm = CGPoint(x: centerXNorm - bottomWidthNorm / 2, y: bottomYNorm)
        let rightBottomNorm = CGPoint(x: centerXNorm + bottomWidthNorm / 2, y: bottomYNorm)

        // Project all corners through aspect-fill transform
        let leftTop = projectNorm(leftTopNorm)
        let rightTop = projectNorm(rightTopNorm)
        let leftBottom = projectNorm(leftBottomNorm)
        let rightBottom = projectNorm(rightBottomNorm)

        leftPath.move(to: leftBottom)
        leftPath.addLine(to: leftTop)

        rightPath.move(to: rightTop)
        rightPath.addLine(to: rightBottom)

        fillPath.move(to: leftTop)
        fillPath.addLine(to: rightTop)
        fillPath.addLine(to: rightBottom)
        fillPath.addLine(to: leftBottom)
        fillPath.closeSubpath()

        return (leftPath, rightPath, fillPath, leftColor, rightColor, showLeftAlert, showRightAlert)
    }
    
    // Clip a point array to y >= minY, interpolating the exact crossing point
    private func clipPointsToY(_ points: [CGPoint], minY: CGFloat) -> [CGPoint] {
        guard points.count >= 2 else { return points }
        var result: [CGPoint] = []
        for i in 0..<points.count - 1 {
            let p1 = points[i]
            let p2 = points[i + 1]
            let y1 = p1.y
            let y2 = p2.y
            
            if y1 >= minY && y2 >= minY {
                // Both below clip line — keep p1 (p2 will be handled next iteration or as last)
                if i == 0 { result.append(p1) }
            } else if y1 < minY && y2 >= minY {
                // Crossing from above to below — add interpolated point at minY
                let t = (minY - y1) / (y2 - y1)
                let x = p1.x + (p2.x - p1.x) * t
                result.append(CGPoint(x: x, y: minY))
            } else if y1 >= minY && y2 < minY {
                // Crossing from below to above — add p1 and interpolated point
                if i == 0 { result.append(p1) }
                let t = (minY - y1) / (y2 - y1)
                let x = p1.x + (p2.x - p1.x) * t
                result.append(CGPoint(x: x, y: minY))
            }
            // Both above — skip
        }
        // Ensure last point if it's below clip line
        if let last = points.last, last.y >= minY, result.last?.x != last.x || result.last?.y != last.y {
            result.append(last)
        }
        return result
    }
}

        let startY = rotCenterY - dirY * halfLen

        let endY = rotCenterY + dirY * halfLen

            path.addLine(to: CGPoint(x: endX, y: endY))

struct DiagnosticEvent: Codable {
    let timestamp: Date
    let type: String
    let speedKMPH: Double?
    let distanceMeters: Double?
    let isDangerous: Bool?
    let orientation: String?
    let details: String?

final class DiagnosticsLogStore {
    private var events: [DiagnosticEvent] = []
    private let lock = NSLock()
    private let encoder: JSONEncoder
    private let maxEvents = 5000
    private let logger = Logger(subsystem: "vn.ai.driveassistant", category: "Diagnostics")

    // Publisher for real-time event observation
    let eventPublisher = PassthroughSubject<DiagnosticEvent, Never>()

    // Daily file logging
    private let dailyLogQueue = DispatchQueue(label: "vn.ai.driveassistant.daily-log", qos: .utility)
    private var currentDailyLogDate: Date?
    private var currentDailyLogHandle: FileHandle?

    init() {
        encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        encoder.outputFormatting = [] // compact JSON Lines
    }

    func record(type: String, speedKMPH: Double? = nil, distanceMeters: Double? = nil, isDangerous: Bool? = nil, orientation: String? = nil, details: String? = nil) {
        let event = DiagnosticEvent(timestamp: Date(), type: type, speedKMPH: speedKMPH, distanceMeters: distanceMeters, isDangerous: isDangerous, orientation: orientation, details: details)
        lock.lock()
        events.append(event)
        if events.count > maxEvents {
            events.removeFirst(events.count - maxEvents)
        }
        lock.unlock()

        // Persist to daily JSONL file asynchronously
        appendToDailyLog(event)

        // Emit event for real-time observers
        eventPublisher.send(event)
    }

    private func appendToDailyLog(_ event: DiagnosticEvent) {
        dailyLogQueue.async { [weak self] in
            self?.writeEventToDailyLog(event)
        }
    }

    private func writeEventToDailyLog(_ event: DiagnosticEvent) {
        let calendar = Calendar.current
        let today = calendar.startOfDay(for: Date())

        // Check if we need to rotate to a new day's file
        if currentDailyLogDate != today {
            closeCurrentDailyLog()
            openDailyLogForDate(today)
        }

        guard let handle = currentDailyLogHandle else { return }

        do {
            let data = try encoder.encode(event)
            var jsonString = String(data: data, encoding: .utf8) ?? ""
            jsonString += "\n"
            if let lineData = jsonString.data(using: .utf8) {
                handle.seekToEndOfFile()
                handle.write(lineData)
            }
        } catch {
            logger.error("Failed to encode daily log event: \(error.localizedDescription, privacy: .public)")
        }
    }

    private func openDailyLogForDate(_ date: Date) {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd"
        let fileName = "DriveAssistant-\(formatter.string(from: date)).jsonl"

        let documentsURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask).first ?? FileManager.default.temporaryDirectory
        let logsDir = documentsURL.appendingPathComponent("DriveAssistantLogs", isDirectory: true)

        do {
            try FileManager.default.createDirectory(at: logsDir, withIntermediateDirectories: true, attributes: nil)
        } catch {
            logger.error("Failed to create logs directory: \(error.localizedDescription, privacy: .public)")
            return
        }

        let fileURL = logsDir.appendingPathComponent(fileName)

        // Create file if it doesn't exist
        if !FileManager.default.fileExists(atPath: fileURL.path) {
            FileManager.default.createFile(atPath: fileURL.path, contents: nil, attributes: nil)
        }

        do {
            currentDailyLogHandle = try FileHandle(forWritingTo: fileURL)
            currentDailyLogDate = date
        } catch {
            logger.error("Failed to open daily log file: \(error.localizedDescription, privacy: .public)")
        }
    }

    private func closeCurrentDailyLog() {
        currentDailyLogHandle?.closeFile()
        currentDailyLogHandle = nil
        currentDailyLogDate = nil
    }

    func clear() {
        lock.lock()
        events.removeAll()
        lock.unlock()

        // Delete all .jsonl files in the logs directory
        dailyLogQueue.async { [weak self] in
            self?.deleteAllDailyLogs()
        }
    }

    private func deleteAllDailyLogs() {
        let documentsURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask).first ?? FileManager.default.temporaryDirectory
        let logsDir = documentsURL.appendingPathComponent("DriveAssistantLogs", isDirectory: true)

        guard FileManager.default.fileExists(atPath: logsDir.path) else { return }

        do {
            let files = try FileManager.default.contentsOfDirectory(at: logsDir, includingPropertiesForKeys: nil)
            for file in files where file.pathExtension == "jsonl" {
                try FileManager.default.removeItem(at: file)
            }
        } catch {
            logger.error("Failed to delete daily logs: \(error.localizedDescription, privacy: .public)")
        }
    }

    func exportURL() -> URL {
        let exportedEvents: [DiagnosticEvent]
        lock.lock()
        exportedEvents = events
        lock.unlock()

        // If no in-memory events, try to read today's daily log file
        let eventsToExport: [DiagnosticEvent]
        if exportedEvents.isEmpty {
            eventsToExport = readTodaysDailyLog()
        } else {
            eventsToExport = exportedEvents
        }

        let payload: [String: AnyEncodable] = [
            "app": AnyEncodable("DriveAssistant"),
            "bundleID": AnyEncodable(Bundle.main.bundleIdentifier ?? "unknown"),
            "systemVersion": AnyEncodable(UIDevice.current.systemVersion),
            "deviceModel": AnyEncodable(UIDevice.current.model),
            "events": AnyEncodable(eventsToExport)
        ]

        // Use pretty-printed encoder for export
        let exportEncoder = JSONEncoder()
        exportEncoder.dateEncodingStrategy = .iso8601
        exportEncoder.outputFormatting = [.prettyPrinted, .sortedKeys]

        let data: Data
        do {
            data = try exportEncoder.encode(payload)
        } catch {
            logger.error("Failed to encode diagnostics: \(error.localizedDescription, privacy: .public)")
            data = Data("{\"error\":\"Unable to encode diagnostics\"}".utf8)
        }

        let documentsURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask).first ?? FileManager.default.temporaryDirectory
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyyMMdd-HHmmss"
        let url = documentsURL.appendingPathComponent("DriveAssistant-Export-\(formatter.string(from: Date())).json")
        do {
            try data.write(to: url, options: .atomic)
        } catch {
            logger.error("Failed to write diagnostics export: \(error.localizedDescription, privacy: .public)")
        }
        return url
    }

    private func readTodaysDailyLog() -> [DiagnosticEvent] {
        let calendar = Calendar.current
        let today = calendar.startOfDay(for: Date())
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd"
        let fileName = "DriveAssistant-\(formatter.string(from: today)).jsonl"

        let documentsURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask).first ?? FileManager.default.temporaryDirectory
        let logsDir = documentsURL.appendingPathComponent("DriveAssistantLogs", isDirectory: true)
        let fileURL = logsDir.appendingPathComponent(fileName)

        guard FileManager.default.fileExists(atPath: fileURL.path) else { return [] }

        do {
            let data = try Data(contentsOf: fileURL)
            let decoder = JSONDecoder()
            decoder.dateDecodingStrategy = .iso8601
            let lines = String(data: data, encoding: .utf8)?.split(separator: "\n") ?? []
            var events: [DiagnosticEvent] = []
            for line in lines {
                if let lineData = line.data(using: .utf8),
                   let event = try? decoder.decode(DiagnosticEvent.self, from: lineData) {
                    events.append(event)
                }
            }
            return events
        } catch {
            logger.error("Failed to read today's daily log: \(error.localizedDescription, privacy: .public)")
            return []
        }
    }

    deinit {
        closeCurrentDailyLog()
    }
}

    private let encodeValue: (Encoder) throws -> Void

    init<T: Encodable>(_ value: T) {
        encodeValue = value.encode
    }

    func encode(to encoder: Encoder) throws {
        try encodeValue(encoder)
    }

    private func applySpeedRateLimit(_ newSpeedKMPH: Double) {
        let now = CACurrentMediaTime()
        if lastSpeedUpdateTime > 0 {
            let dt = now - lastSpeedUpdateTime
            if dt > 0 {
                let maxDeltaKMPH = maxAccelerationMPS2 * dt * 3.6
                let delta = newSpeedKMPH - lastSpeedKMPH
                if abs(delta) > maxDeltaKMPH {
                    let clampedDelta = delta > 0 ? maxDeltaKMPH : -maxDeltaKMPH
                    speedKMPH = lastSpeedKMPH + clampedDelta
                    lastSpeedUpdateTime = now
                    lastSpeedKMPH = speedKMPH
                    return
                }
            }
        }
        speedKMPH = newSpeedKMPH
        lastSpeedKMPH = newSpeedKMPH
        lastSpeedUpdateTime = now
    }

    func cycleRoadCondition() {
        switch roadCondition {
        case .auto: roadCondition = .dry
        case .dry: roadCondition = .wet
        case .wet: roadCondition = .auto

    private func updateDistanceHistory(distance: Double) {
        let now = CACurrentMediaTime()
        distanceHistory.append((now, distance))
        let cutoff = now - closingSpeedWindow
        distanceHistory.removeAll { $0.timestamp < cutoff }

        guard let oldest = distanceHistory.first,
              now - oldest.timestamp > 0 else {
            closingSpeedMPS = 0
            timeToCollision = nil
            return
        }
        let elapsed = now - oldest.timestamp
        let closingSpeed = max(0, (oldest.distance - distance) / elapsed)
        closingSpeedMPS = min(closingSpeed, maxClosingSpeedMPS)
        timeToCollision = closingSpeedMPS > 0.5 ? distance / closingSpeedMPS : nil

        // Oncoming closing speed spike diagnostic: closing speed exceeds ego speed by >8 m/s
        // while speed >= 15 km/h (indicates likely oncoming traffic false positive)
        if speedKMPH >= 15 && closingSpeedMPS > (speedKMPH / 3.6) + 8.0 {
            diagnostics.record(type: "oncoming_closing_speed_spike",
                details: "speedKMPH=\(String(format: "%.1f", speedKMPH)), closingSpeedMPS=\(String(format: "%.1f", closingSpeedMPS)), distanceM=\(String(format: "%.1f", distance)), leadBox=\(camera.leadVehicleBox?.debugDescription ?? "nil")")
        }
    }

    private func updateAlertState() {
        guard let distance else {
            criticalFrameCount = 0
            setAlertState(.normal)
            return
        }

        let alertsSuppressed = distanceRuleMode == .vnLaw && speedKMPH < 10
        let isLowSpeedWarning = distanceRuleMode == .timeHeadway && speedKMPH < 10 && distance < 4
        let isWarning = !alertsSuppressed && speedKMPH >= 10 && distance < activeMinimumDistance
        // TTC-based critical only above 25 km/h; below 25 km/h use absolute distance < 3.5m
        let isCriticalCandidate = (speedKMPH >= 25.0 && (timeToCollision.map { $0 < 1.8 } ?? false)) ||
            (speedKMPH < 25.0 && distance < 3.5)

        if isCriticalCandidate {
            criticalFrameCount += 1
        } else {
            criticalFrameCount = 0
        }

        if alertsSuppressed {
            criticalFrameCount = 0
            setAlertState(.normal)
        } else if criticalFrameCount >= 3 {
            setAlertState(.critical)
            alerts.playCriticalAlert(speak: voiceAlertsEnabled, reduceSpeed: timeToCollision.map { $0 < 1.8 } ?? false)
        } else if isLowSpeedWarning || isWarning {
            setAlertState(.warning)
            if speedKMPH >= 10 {
                alerts.playWarningAlert()
            }
        } else {
            setAlertState(.normal)
        }

        // Stop-and-Go alert logic
        updateStopAndGoAlert(distance: distance)
    }

    private func setAlertState(_ state: ADASAlertState) {
        guard alertState != state else {
            isDangerous = state == .critical
            return
        }
        alertState = state
        isDangerous = state == .critical
        diagnostics.record(type: "alert_state_changed", speedKMPH: speedKMPH, distanceMeters: distance, isDangerous: isDangerous, orientation: String(describing: UIDevice.current.orientation), details: "state=\(String(describing: state)), closingSpeedMPS=\(closingSpeedMPS), ttc=\(timeToCollision.map { String($0) } ?? "nil")")
    }

    /// Stop-and-Go (Lead Vehicle Departure) alert
    private func updateStopAndGoAlert(distance: Double) {
        let now = CACurrentMediaTime()

        // Disarm if speed >= 5 km/h (moving)
        if speedKMPH >= 5.0 {
            if stopAndGoArmed {
                stopAndGoArmed = false
                stoppedLeadDistance = 0
                leadVehicleStableStartTime = nil
                departureDetectedStartTime = nil
            }
            leadVehicleGoneStartTime = nil
            return
        }

        // Disarm if lead vehicle gone for > 1.5s (distance > 20m means no lead vehicle)
        if distance > 20.0 {
            if leadVehicleGoneStartTime == nil {
                leadVehicleGoneStartTime = now
            } else if now - leadVehicleGoneStartTime! >= 1.5 {
                if stopAndGoArmed {
                    stopAndGoArmed = false
                    stoppedLeadDistance = 0
                    leadVehicleStableStartTime = nil
                    departureDetectedStartTime = nil
                }
                leadVehicleGoneStartTime = nil
            }
            return
        } else {
            leadVehicleGoneStartTime = nil
        }

        // Arm: stopped (speed < 3), lead vehicle at 1.5-9m for >= 2.5s
        let hasLeadVehicleInRange = distance >= 1.5 && distance <= 9.0
        if stopAndGoAlertEnabled && speedKMPH < 3.0 && hasLeadVehicleInRange {
            if leadVehicleStableStartTime == nil {
                leadVehicleStableStartTime = now
            } else if now - leadVehicleStableStartTime! >= 2.5 && !stopAndGoArmed {
                stopAndGoArmed = true
                stoppedLeadDistance = distance  // minimum distance while stopped
                departureDetectedStartTime = nil
            }
        } else {
            leadVehicleStableStartTime = nil
        }

        // While armed and stopped, watch for distance increase >= 1.8m above stoppedLeadDistance
        if stopAndGoArmed && speedKMPH < 3.0 && closingSpeedMPS == 0 {
            let distanceIncrease = distance - stoppedLeadDistance
            if distanceIncrease >= 1.8 {
                if departureDetectedStartTime == nil {
                    departureDetectedStartTime = now
                } else if now - departureDetectedStartTime! >= 0.4 {
                    // Trigger alert (respect cooldown)
                    if now - lastStopAndGoAlertTime >= stopAndGoAlertCooldown {
                        alerts.playStopAndGoAlert(speak: voiceAlertsEnabled)
                        lastStopAndGoAlertTime = now
                        isLeadVehicleMovingAlertActive = true

                        // Auto-hide after 3s
                        DispatchQueue.main.asyncAfter(deadline: .now() + 3.0) { [weak self] in
                            self?.isLeadVehicleMovingAlertActive = false
                        }
                    }
                    // Disarm after trigger (only once per stop)
                    stopAndGoArmed = false
                    stoppedLeadDistance = 0
                    leadVehicleStableStartTime = nil
                    departureDetectedStartTime = nil
                }
            } else {
                departureDetectedStartTime = nil
                // Update stoppedLeadDistance to track minimum distance while stopped
                if distance < stoppedLeadDistance || stoppedLeadDistance == 0 {
                    stoppedLeadDistance = distance
                }
            }
        } else {
            departureDetectedStartTime = nil
        }
    }

    /// Overspeed alert based on detected speed limit sign
    private func updateOverspeedState() {
        guard enableSpeedSignRecognition else {
            isOverspeeding = false
            return
        }
        guard let limit = detectedSpeedLimit else {
            isOverspeeding = false
            return
        }
        let newOverspeeding = speedKMPH > Double(limit) + overspeedToleranceKmh
        if newOverspeeding != isOverspeeding {
            isOverspeeding = newOverspeeding
            if isOverspeeding {
                alerts.playOverspeedAlert(limit: limit, speak: voiceAlertsEnabled)
                diagnostics.record(type: "overspeed_alert", details: "speedKMPH=\(String(format: "%.1f", speedKMPH)), limit=\(limit), toleranceKmh=\(String(format: "%.1f", overspeedToleranceKmh))")
            }
        }
    }

    /// Apply Vietnam TT38 statutory speed limits based on residential area + divided road status.
    /// Called when isInResidentialArea or isDividedRoad changes, or when explicit P.127 sign clears.
    /// - If explicit P.127 sign is active (isCurrentLimitStatutory == false), do nothing.
    /// - If statutory limit is active (isCurrentLimitStatutory == true) or no limit set, update to new TT38 value.
    ///     * isInResidentialArea == true:  50 km/h (2-way) or 60 km/h (divided)
    ///     * isInResidentialArea == false: 80 km/h (2-way) or 90 km/h (divided)  [if enableDefaultOutsideResidentialLimit]
    ///     * isInResidentialArea == nil:   do nothing (unknown)
    private func applyTT38StatutoryLimit() {
        // If an explicit P.127 speed sign was confirmed by camera, respect it and don't override
        guard !isCurrentLimitStatutory || detectedSpeedLimit == nil else { return }
        guard let inResidential = isInResidentialArea else { return }  // unknown state
        guard enableDefaultOutsideResidentialLimit || inResidential else { return }  // outside residential but feature disabled

        let newLimit: Int
        if inResidential {
            newLimit = isDividedRoad ? 60 : 50
        } else {
            newLimit = isDividedRoad ? 90 : 80
        }

        // Only update if limit actually changed (or was nil)
        if detectedSpeedLimit != newLimit {
            detectedSpeedLimit = newLimit
            isCurrentLimitStatutory = true
            diagnostics.record(type: "tsr_tt38_statutory", details: "limit=\(newLimit) inResidential=\(inResidential) divided=\(isDividedRoad) source=\(isDividedRoad ? (camera.trafficSignDetector.isDividedRoadFromSign ? "sign" : "osm") : "osm")")
            updateOverspeedState()
        }
    }

    /// Dynamic in-drive horizon auto-calibration with progress tracking
    private func attemptHorizonAutoCalibration(box: CGRect?, confidence: Float) {
        // Check for mount shift on every frame
        motion.checkMountShift(speedKMPH: speedKMPH)

        // Skip calibration if mount is shifted or already auto-calibrated
        guard showHorizonGuide,
              motion.isActive,
              speedKMPH > 25.0,
              abs(motion.deltaRoll) < 2.0,
              !motion.isMountShifted else {
            horizonCalibrationFrames = 0
            lastTrackedBox = nil
            return
        }

        // Use lead vehicle box or confident lane lines for stable tracking
        let hasStableLeadVehicle = box != nil && confidence >= 0.70 && box!.width >= 0.05 && box!.width <= 0.25
        let hasConfidentLanes = laneDetector.isLaneDetectionActive &&
                                laneDetector.leftLaneConfidence > 0.6 &&
                                laneDetector.rightLaneConfidence > 0.6 &&
                                !laneDetector.leftLanePoints.isEmpty &&
                                !laneDetector.rightLanePoints.isEmpty
        let hasStableTracking = hasStableLeadVehicle || hasConfidentLanes

        if hasStableLeadVehicle, let box = box {
            if let lastBox = lastTrackedBox {
                let iou = intersectionOverUnion(lastBox, box)
                if iou < 0.5 {
                    horizonCalibrationFrames = 0
                    lastTrackedBox = box
                    // Still record calibration frame for lane-based stability
                    motion.recordCalibrationFrame(speedKMPH: speedKMPH, hasStableTracking: hasConfidentLanes, horizonGuideEnabled: showHorizonGuide)
                    return
                }
            }
            lastTrackedBox = box
            horizonCalibrationFrames += 1
        } else if hasConfidentLanes {
            // Lane-based calibration frame
            horizonCalibrationFrames += 1
        } else {
            horizonCalibrationFrames = 0
            lastTrackedBox = nil
            motion.recordCalibrationFrame(speedKMPH: speedKMPH, hasStableTracking: false, horizonGuideEnabled: showHorizonGuide)
            return
        }

        // Record calibration progress
        motion.recordCalibrationFrame(speedKMPH: speedKMPH, hasStableTracking: hasStableTracking, horizonGuideEnabled: showHorizonGuide)

        guard horizonCalibrationFrames >= 5 else { return }

        // Infer horizon from lead vehicle or lane geometry
        var inferredHorizonY: Double = 0.45
        if hasStableLeadVehicle, let box = box {
            inferredHorizonY = (1.0 - box.minY) - (box.height * 0.15)
        } else if hasConfidentLanes,
                  let laneEdges = laneDetector.getLaneEdgesAtY(0.85, imageSize: camera.cameraFrameSize) {
            // Use lane geometry to infer horizon (vanishing point)
            let laneCenter = (laneEdges.left + laneEdges.right) / 2.0
            inferredHorizonY = 0.45 + (laneCenter - 0.5) * 0.1  // Small adjustment
        }

        let clampedInferred = max(0.25, min(0.65, inferredHorizonY))
        motion.updateEffectiveHorizonFromVision(inferredHorizonY: clampedInferred)
    }
    
    private func intersectionOverUnion(_ lhs: CGRect, _ rhs: CGRect) -> CGFloat {
        let intersection = lhs.intersection(rhs)
        guard !intersection.isNull, intersection.width > 0, intersection.height > 0 else { return 0 }
        let union = lhs.width * lhs.height + rhs.width * rhs.height - intersection.width * intersection.height
        return union > 0 ? intersection.width * intersection.height / union : 0
    }
}

    var area: CGFloat { width * height }

    private func fitQuadratic(points: [CGPoint], imageSize: CGSize, isLeft: Bool) -> [CGPoint]? {
        guard points.count >= 10 else { return nil }
        
        var sumY = 0.0, sumY2 = 0.0, sumY3 = 0.0, sumY4 = 0.0
        var sumX = 0.0, sumXY = 0.0, sumXY2 = 0.0
        
        for point in points {
            let y = point.y
            let x = point.x
            sumY += y
            sumY2 += y * y
            sumY3 += y * y * y
            sumY4 += y * y * y * y
            sumX += x
            sumXY += x * y
            sumXY2 += x * y * y
        }
        
        let n = Double(points.count)
        
        let A = [
            [sumY4, sumY3, sumY2],
            [sumY3, sumY2, sumY],
            [sumY2, sumY, n]
        ]
        let B = [sumXY2, sumXY, sumX]
        
        guard let coeffs = solveLinearSystem(A, B) else { return nil }
        
        let a = coeffs[0], b = coeffs[1], c = coeffs[2]
        
        var fittedPoints: [CGPoint] = []
        let startY = roiBottomY
        let endY = roiTopY
        let steps = 20
        
        for i in 0..<steps {
            let y = startY - (startY - endY) * Double(i) / Double(steps - 1)
            let x = a * y * y + b * y + c
            let clampedX = max(0.0, min(1.0, x))
            fittedPoints.append(CGPoint(x: clampedX, y: y))
        }
        
        let centerX = 0.5 + roiCenterOffsetX
        let avgX = fittedPoints.map { $0.x }.reduce(0, +) / Double(fittedPoints.count)
        
        if isLeft && avgX > centerX { return nil }
        if !isLeft && avgX < centerX { return nil }
        
        return fittedPoints
    }
    
    private func solveLinearSystem(_ A: [[Double]], _ B: [Double]) -> [Double]? {
        var M = A
        var b = B
        
        for i in 0..<3 {
            var maxRow = i
            for j in (i+1)..<3 {
                if abs(M[j][i]) > abs(M[maxRow][i]) { maxRow = j }
            }
            
            if abs(M[maxRow][i]) < 1e-10 { return nil }
            
            if maxRow != i {
                M.swapAt(i, maxRow)
                b.swapAt(i, maxRow)
            }
            
            let pivot = M[i][i]
            for j in i..<3 { M[i][j] /= pivot }
            b[i] /= pivot
            
            for j in (i+1)..<3 {
                let factor = M[j][i]
                for k in i..<3 { M[j][k] -= factor * M[i][k] }
                b[j] -= factor * b[i]
            }
        }
        
        var x = [0.0, 0.0, 0.0]
        for i in (0..<3).reversed() {
            x[i] = b[i]
            for j in (i+1)..<3 { x[i] -= M[i][j] * x[j] }
        }
        
        return x
    }
    
    // MARK: - EMA with Grace Window
    private func applyEMAWithGrace(leftFit: [CGPoint]?, rightFit: [CGPoint]?) -> (left: [CGPoint], right: [CGPoint], leftConf: Float, rightConf: Float) {
        var leftConf: Float = 0
        var rightConf: Float = 0
        var smoothedLeft = emaLeftLane ?? []
        var smoothedRight = emaRightLane ?? []
        
        // LEFT lane
        if let leftFit {
            leftConf = 0.8
            framesSinceConfidentLeft = 0
            if emaLeftLane == nil {
                emaLeftLane = leftFit
                smoothedLeft = leftFit
            } else {
                let minCount = min(emaLeftLane!.count, leftFit.count)
                for i in 0..<minCount {
                    emaLeftLane![i].x = emaAlpha * leftFit[i].x + (1 - emaAlpha) * emaLeftLane![i].x
                    emaLeftLane![i].y = emaAlpha * leftFit[i].y + (1 - emaAlpha) * emaLeftLane![i].y
                }
                smoothedLeft = emaLeftLane!
            }
        } else {
            framesSinceConfidentLeft += 1
            if framesSinceConfidentLeft <= graceFrames {
                // Grace period: hold last EMA with decaying confidence
                leftConf = max(0.5 - Float(framesSinceConfidentLeft) * 0.5 / Float(graceFrames), 0.1)
                smoothedLeft = emaLeftLane ?? []
            } else {
                // Fade to static fallback
                leftConf = 0
                emaLeftLane = nil
                smoothedLeft = []
            }
        }
        
        // RIGHT lane
        if let rightFit {
            rightConf = 0.8
            framesSinceConfidentRight = 0
            if emaRightLane == nil {
                emaRightLane = rightFit
                smoothedRight = rightFit
            } else {
                let minCount = min(emaRightLane!.count, rightFit.count)
                for i in 0..<minCount {
                    emaRightLane![i].x = emaAlpha * rightFit[i].x + (1 - emaAlpha) * emaRightLane![i].x
                    emaRightLane![i].y = emaAlpha * rightFit[i].y + (1 - emaAlpha) * emaRightLane![i].y
                }
                smoothedRight = emaRightLane!
            }
        } else {
            framesSinceConfidentRight += 1
            if framesSinceConfidentRight <= graceFrames {
                rightConf = max(0.5 - Float(framesSinceConfidentRight) * 0.5 / Float(graceFrames), 0.1)
                smoothedRight = emaRightLane ?? []
            } else {
                rightConf = 0
                emaRightLane = nil
                smoothedRight = []
            }
        }
        
        return (smoothedLeft, smoothedRight, leftConf, rightConf)
    }
    
    // MARK: - Lane Departure Detection (Anti-False-Positive)
    private func checkLaneDeparture(speedKMPH: Double, vehicleCenterX: Double) {
        let now = CACurrentMediaTime()

        // Reset if LDWS disabled or speed gated
        if !isLDWSEnabled || speedKMPH < minSpeedForDepartureAlert {
            leftDepartureStartTime = nil
            rightDepartureStartTime = nil
            departureState = .none
            departureAlertActive = false
            lastDepartureAlertActive = false
            return
        }

        // Require both lanes with good confidence
        guard leftLaneConfidence > 0.5, rightLaneConfidence > 0.5,
              !leftLanePoints.isEmpty, !rightLanePoints.isEmpty else {
            leftDepartureStartTime = nil
            rightDepartureStartTime = nil
            departureState = .none
            return
        }

        // Get lane edges at bottom of ROI (normalized coordinates)
        let bottomLeftX = leftLanePoints.last?.x ?? 0
        let bottomRightX = rightLanePoints.last?.x ?? 1.0
        let laneWidth = bottomRightX - bottomLeftX
        let thresholdRatio = departureThresholdRatio  // 8% of lane width

        // LEFT departure: vehicle within 8% of left edge OR crossing it
        let leftDistance = vehicleCenterX - bottomLeftX
        let isNearLeftEdge = leftDistance < thresholdRatio && leftDistance >= 0
        let hasCrossedLeft = leftDistance < 0

        if hasCrossedLeft || isNearLeftEdge {
            if leftDepartureStartTime == nil { leftDepartureStartTime = now }
            let duration = now - leftDepartureStartTime!

            if hasCrossedLeft {
                departureState = .driftingLeft
            } else if duration >= departurePersistenceThreshold {
                departureState = .driftingLeft
            }

            // Trigger alert only on actual crossing with persistence + cooldown
            if hasCrossedLeft && duration >= departurePersistenceThreshold && now - lastDepartureAlertTime >= departureCooldown {
                departureAlertActive = true
                lastDepartureAlertTime = now
            }
        } else {
            leftDepartureStartTime = nil
            if departureState == .driftingLeft { departureState = .none }
        }

        // RIGHT departure: vehicle within 8% of right edge OR crossing it
        let rightDistance = bottomRightX - vehicleCenterX
        let isNearRightEdge = rightDistance < thresholdRatio && rightDistance >= 0
        let hasCrossedRight = rightDistance < 0

        if hasCrossedRight || isNearRightEdge {
            if rightDepartureStartTime == nil { rightDepartureStartTime = now }
            let duration = now - rightDepartureStartTime!

            if hasCrossedRight {
                departureState = .driftingRight
            } else if duration >= departurePersistenceThreshold {
                departureState = .driftingRight
            }

            if hasCrossedRight && duration >= departurePersistenceThreshold && now - lastDepartureAlertTime >= departureCooldown {
                departureAlertActive = true
                lastDepartureAlertTime = now
            }
        } else {
            rightDepartureStartTime = nil
            if departureState == .driftingRight { departureState = .none }
        }

        // Clear alert after cooldown
        if departureAlertActive && now - lastDepartureAlertTime >= departureCooldown {
            departureAlertActive = false
        }

        // Log departure alert on rising edge
        if departureAlertActive && !lastDepartureAlertActive {
            diagnostics?.record(type: "ldws_departure_alert",
                details: "departureState=\(String(describing: departureState)), speedKMPH=\(String(format: "%.1f", speedKMPH)), leftConf=\(String(format: "%.2f", leftLaneConfidence)), rightConf=\(String(format: "%.2f", rightLaneConfidence))")
        }
        lastDepartureAlertActive = departureAlertActive
    }
    
    /// Get dynamic ROI polygon that snaps to detected lanes (for vehicle filtering)
    func getDynamicROIPolygon(imageSize: CGSize) -> [CGPoint]? {
        guard leftLaneConfidence > 0.5, rightLaneConfidence > 0.5,
              !leftLanePoints.isEmpty, !rightLanePoints.isEmpty else {
            return nil
        }
        
        let topLeft = leftLanePoints.first ?? CGPoint(x: 0.39, y: roiTopY)
        let topRight = rightLanePoints.first ?? CGPoint(x: 0.61, y: roiTopY)
        let bottomLeft = leftLanePoints.last ?? CGPoint(x: 0.26, y: roiBottomY)
        let bottomRight = rightLanePoints.last ?? CGPoint(x: 0.74, y: roiBottomY)
        
        return [topLeft, topRight, bottomRight, bottomLeft]
    }
    
    /// Get normalized lane edges at specific Y (for vehicle filter)
    func getLaneEdgesAtY(_ y: CGFloat, imageSize: CGSize) -> (left: CGFloat, right: CGFloat)? {
        guard leftLaneConfidence > 0.5, rightLaneConfidence > 0.5 else { return nil }
        
        let leftX = interpolateLaneAtY(leftLanePoints, targetY: y)
        let rightX = interpolateLaneAtY(rightLanePoints, targetY: y)
        
        guard let leftX, let rightX, leftX < rightX else { return nil }
        return (leftX, rightX)
    }
    
    private func interpolateLaneAtY(_ points: [CGPoint], targetY: CGFloat) -> CGFloat? {
        guard points.count >= 2 else { return nil }
        
        for i in 1..<points.count {
            let p1 = points[i-1]
            let p2 = points[i]
            
            if (p1.y <= targetY && p2.y >= targetY) || (p1.y >= targetY && p2.y <= targetY) {
                let t = (targetY - p1.y) / (p2.y - p1.y)
                return p1.x + (p2.x - p1.x) * t
            }
        }
        
        if targetY > points.last!.y {
            let p1 = points[points.count-2]
            let p2 = points.last!
            let t = (targetY - p1.y) / (p2.y - p1.y)
            return p1.x + (p2.x - p1.x) * t
        }
        
        return points.first?.x
    }
}

final class LocationManager: NSObject, ObservableObject, CLLocationManagerDelegate {

        super.init()
        manager.delegate = self
        manager.desiredAccuracy = kCLLocationAccuracyBest
        manager.distanceFilter = kCLDistanceFilterNone

        manager.requestWhenInUseAuthorization()
        manager.startUpdatingLocation()
        startStallTimer()
    }

        manager.stopUpdatingLocation()
        stallTimer?.invalidate()
        stallTimer = nil
    }

    private func startStallTimer() {
        stallTimer?.invalidate()
        stallTimer = Timer.scheduledTimer(withTimeInterval: stallThreshold, repeats: true) { [weak self] _ in
            self?.handleStall()
        }
    }

    private func handleStall() {
        DispatchQueue.main.async { [weak self] in
            self?.speedKMPH = 0
            self?.horizontalAccuracy = .infinity
        }
    }

    func locationManager(_ manager: CLLocationManager, didUpdateLocations locations: [CLLocation]) {
        guard let location = locations.last else { return }
        horizontalAccuracy = location.horizontalAccuracy
        guard location.speed >= 0,
              location.horizontalAccuracy >= 0,
              location.horizontalAccuracy <= 50 else {
            return
        }
        speedKMPH = location.speed * 3.6
        startStallTimer()
        computeIsAstronomicalNight(for: location)
        weatherService.update(for: location) { [weak self] condition in
            self?.weatherRoadCondition = condition
        }
        // Query OSM road metadata (throttled) - call on MainActor
        Task { @MainActor in
            roadMetadata.maybeQueryRoadMetadata(for: location)
        }
    }

    /// Compute whether sun is below horizon using NOAA solar elevation formula
    private func computeIsAstronomicalNight(for location: CLLocation) {
        var utcCal = Calendar(identifier: .gregorian)
        utcCal.timeZone = TimeZone(secondsFromGMT: 0)!
        let date = Date()
        let year = utcCal.component(.year, from: date)
        let month = utcCal.component(.month, from: date)
        let day = utcCal.component(.day, from: date)
        let hour = utcCal.component(.hour, from: date)
        let minute = utcCal.component(.minute, from: date)
        let second = utcCal.component(.second, from: date)
        
        // Fractional day of year (1-365/366)
        var dayOfYear = 0
        for m in 1..<month {
            dayOfYear += daysInMonth(month: m, year: year)
        }
        dayOfYear += day
        let dayFraction = Double(hour) / 24.0 + Double(minute) / 1440.0 + Double(second) / 86400.0
        
        // Julian day number
        let jd = Double(dayOfYear) + dayFraction + 2451545.0 - 0.5 + Double(year - 2000) * 365.25
        
        // Julian centuries since J2000
        let n = jd - 2451545.0
        let L = (280.460 + 0.9856474 * n).truncatingRemainder(dividingBy: 360.0)
        let g = (357.528 + 0.9856003 * n).truncatingRemainder(dividingBy: 360.0) * .pi / 180.0
        let lambda = (L + 1.915 * sin(g) + 0.020 * sin(2 * g)) * .pi / 180.0
        let epsilon = (23.439 - 0.0000004 * n) * .pi / 180.0
        
        // Right ascension and declination
        let sinDelta = sin(epsilon) * sin(lambda)
        let delta = asin(sinDelta)  // Solar declination
        
        // Hour angle
        let lat = location.coordinate.latitude * .pi / 180.0
        let lon = location.coordinate.longitude * .pi / 180.0
        let gmst = (280.46061837 + 360.98564736629 * n) * .pi / 180.0
        let lst = gmst + lon
        let hourAngle = lst - atan2(sin(lambda) * cos(epsilon), cos(lambda))
        
        // Solar elevation angle
        let sinElev = sin(lat) * sin(delta) + cos(lat) * cos(delta) * cos(hourAngle)
        let elevation = asin(max(-1.0, min(1.0, sinElev))) * 180.0 / .pi
        
        // Civil twilight zenith: 90.833° (accounts for atmospheric refraction + sun disc)
        let isNight = elevation < -0.833  // Below horizon (civil twilight)
        
        DispatchQueue.main.async { [weak self] in
            self?.isAstronomicalNight = isNight
        }
    }
    
    private func daysInMonth(month: Int, year: Int) -> Int {
        switch month {
        case 1, 3, 5, 7, 8, 10, 12: return 31
        case 4, 6, 9, 11: return 30
        case 2:
            return (year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)) ? 29 : 28
        default: return 30
        }
    }
}

final class MotionManager: ObservableObject {
    private let motionManager = CMMotionManager()
    private let updateInterval = 0.1  // 10 Hz

    // Raw gyroscope values
    @Published var rollDegrees: Double = 0
    @Published var pitchDegrees: Double = 0
    @Published var horizonY: Double = 0.5
    @Published var isActive: Bool = false
    @Published var isLevel: Bool = false

    // Reference mount calibration
    @AppStorage("calibratedPitch") var calibratedPitch: Double = 0.0
    @AppStorage("calibratedRoll") var calibratedRoll: Double = 0.0
    @AppStorage("hasSavedMountProfile") var hasSavedMountProfile: Bool = false

    // Learned horizon offset from auto-calibration (persisted)
    @AppStorage("learnedHorizonOffset") private var learnedHorizonOffset: Double = 0.0

    // Delta from calibrated reference
    @Published var deltaPitch: Double = 0
    @Published var deltaRoll: Double = 0

    // Dynamic effective horizon (gyro baseline + in-drive EMA)
    @Published var effectiveHorizonY: Double = 0.45

    // Mount alignment state
    @Published var isMountAligned: Bool = false

    // Auto-calibration progress (0...100)
    @Published var calibrationProgress: Int = 0
    @Published var isAutoCalibrated: Bool = false

    // Mount shift detection
    @Published var isMountShifted: Bool = false
    private var mountShiftStartTime: TimeInterval?
    private let mountShiftThreshold: TimeInterval = 2.0  // seconds
    private let mountShiftAngleThreshold: Double = 3.5  // degrees

    private let verticalFOV: Double = 41.0  // iPhone 1x landscape vertical FOV
    private let baseHorizonY: Double = 0.45  // Base normalized horizon (45% from top)
    private let horizonYRange: ClosedRange<Double> = 0.25...0.65
    private let autoCalibrateAlpha: Double = 0.02  // Slow EMA for in-drive calibration
    private let autoCalibrateClamp: Double = 0.06  // Max deviation from gyro baseline

    init() {
        // Initialize effectiveHorizonY from stored calibration
        computeDeltasAndHorizon()

        // Initialize calibration state from persisted profile
        if hasSavedMountProfile {
            calibrationProgress = 100
            isAutoCalibrated = true
            effectiveHorizonY = horizonY + learnedHorizonOffset
        }
    }
    
    func saveCurrentMountAngle() {
        calibratedPitch = pitchDegrees
        calibratedRoll = rollDegrees
        hasSavedMountProfile = true
        calibrationProgress = 100
        isAutoCalibrated = true
        learnedHorizonOffset = effectiveHorizonY - horizonY
        computeDeltasAndHorizon()
    }

    func resetMountProfile() {
        calibratedPitch = 0.0
        calibratedRoll = 0.0
        hasSavedMountProfile = false
        learnedHorizonOffset = 0.0
        calibrationProgress = 0
        isAutoCalibrated = false
        isMountShifted = false
        mountShiftStartTime = nil
        computeDeltasAndHorizon()
    }

    /// Call from DriveAssistantModel when a valid calibration frame is processed during driving
    func recordCalibrationFrame(speedKMPH: Double, hasStableTracking: Bool, horizonGuideEnabled: Bool) {
        guard horizonGuideEnabled && isActive && speedKMPH > 25.0 && abs(deltaRoll) < 2.0 && hasStableTracking else {
            return
        }

        // Increment progress by ~2% per valid frame
        calibrationProgress = min(calibrationProgress + 2, 100)

        if calibrationProgress >= 100 && !isAutoCalibrated {
            isAutoCalibrated = true
            // Save current pose as the new calibrated reference so deltas are measured against it
            calibratedPitch = pitchDegrees
            calibratedRoll = rollDegrees
            hasSavedMountProfile = true
            // Persist learned horizon offset (measured against new reference)
            learnedHorizonOffset = effectiveHorizonY - horizonY
            isMountShifted = false
            mountShiftStartTime = nil
            // Recompute deltas with new reference
            computeDeltasAndHorizon()
        }
    }

    /// Call from DriveAssistantModel to check for mount shift
    func checkMountShift(speedKMPH: Double) {
        let hasProfile = hasSavedMountProfile || isAutoCalibrated
        let isShiftedNow = hasProfile && (abs(deltaRoll) > mountShiftAngleThreshold || abs(deltaPitch) > mountShiftAngleThreshold) && speedKMPH > 10.0

        let now = CACurrentMediaTime()
        if isShiftedNow {
            if mountShiftStartTime == nil {
                mountShiftStartTime = now
            } else if now - mountShiftStartTime! >= mountShiftThreshold {
                if !isMountShifted {
                    isMountShifted = true
                    // Reset calibration progress to force re-calibration once mount settles
                    calibrationProgress = 0
                    isAutoCalibrated = false
                }
            }
        } else {
            mountShiftStartTime = nil
            // Auto-clear mount shift when alignment is restored
            if isMountShifted && hasProfile && abs(deltaRoll) <= 1.5 && abs(deltaPitch) <= 1.5 {
                isMountShifted = false
            }
        }
    }
    
    private func computeDeltasAndHorizon() {
        deltaPitch = pitchDegrees - calibratedPitch
        deltaRoll = rollDegrees - calibratedRoll

        // Gyro-based horizon from delta pitch
        var hY = baseHorizonY + (deltaPitch / verticalFOV)
        hY = max(horizonYRange.lowerBound, min(horizonYRange.upperBound, hY))
        horizonY = hY

        // Preserve learned horizon offset if auto-calibrated and not mount-shifted
        if isAutoCalibrated && !isMountShifted {
            effectiveHorizonY = hY + learnedHorizonOffset
            effectiveHorizonY = max(horizonYRange.lowerBound, min(horizonYRange.upperBound, effectiveHorizonY))
        } else {
            effectiveHorizonY = hY
        }

        // Mount aligned: within ±1.5° of saved reference
        let hasProfile = hasSavedMountProfile || isAutoCalibrated
        isMountAligned = hasProfile && abs(deltaRoll) <= 1.5 && abs(deltaPitch) <= 1.5

        // Level check: roll within ±1.5° and pitch within ±3.0° (relative to calibrated)
        isLevel = abs(deltaRoll) <= 1.5 && abs(deltaPitch) <= 3.0
    }
    
    func start() {
        guard motionManager.isDeviceMotionAvailable else {
            print("MotionManager: Device motion not available")
            return
        }
        
        motionManager.deviceMotionUpdateInterval = updateInterval
        motionManager.startDeviceMotionUpdates(using: .xArbitraryZVertical, to: .main) { [weak self] motion, error in
            guard let self, let motion = motion else { return }
            
            if let error = error {
                print("MotionManager: Device motion error: \(error)")
                return
            }
            
            // Use gravity vector to compute attitude (Gimbal Lock free for landscape mounting)
            self.updateAttitudeFromGravity(motion.gravity)
        }
        isActive = true
    }
    
    /// Call from DriveAssistantModel when driving straight with stable lead vehicle
    func updateEffectiveHorizonFromVision(inferredHorizonY: Double) {
        // Slow EMA toward vision-inferred horizon
        let newEffective = effectiveHorizonY * (1 - autoCalibrateAlpha) + inferredHorizonY * autoCalibrateAlpha

        // Clamp within ±0.06 of gyro baseline
        let minAllowed = horizonY - autoCalibrateClamp
        let maxAllowed = horizonY + autoCalibrateClamp
        effectiveHorizonY = max(minAllowed, min(maxAllowed, newEffective))

        // Persist learned offset continuously during adaptation
        if isAutoCalibrated && !isMountShifted {
            learnedHorizonOffset = effectiveHorizonY - horizonY
        }
    }
    
    func stop() {
        motionManager.stopDeviceMotionUpdates()
        isActive = false
    }
    
    deinit {
        stop()
    }
}

// MARK: - Gravity-vector attitude computation (Gimbal Lock free)
extension MotionManager {
    /// Update roll/pitch from gravity vector. Call on each device motion update.
    /// Uses gravity vector directly - works in any orientation without Gimbal Lock.
    func updateAttitudeFromGravity(_ gravity: CMAcceleration) {
        let gx = gravity.x
        let gy = gravity.y
        let gz = gravity.z
        
        // Roll: lateral tilt (left/right horizon tilt)
        // gx <= 0: notch left (landscapeLeft) -> atan2(-gy, -gx)
        // gx > 0: notch right (landscapeRight) -> atan2(gy, gx)
        // Both give 0° when iPhone is held upright in landscape
        let rawRoll: Double
        if gx <= 0 {
            rawRoll = atan2(-gy, -gx) * 180.0 / .pi
        } else {
            rawRoll = atan2(gy, gx) * 180.0 / .pi
        }
        
        // Pitch: vertical camera elevation (0 when horizontal/level to road)
        // asin(clamp(gz)) gives angle from horizontal plane
        // Positive pitch = tilted up toward sky, negative = tilted down toward hood
        let rawPitch = asin(max(-1.0, min(1.0, gz))) * 180.0 / .pi
        
        // Smooth with EMA
        let alpha = 0.3
        rollDegrees = rollDegrees * (1 - alpha) + rawRoll * alpha
        pitchDegrees = pitchDegrees * (1 - alpha) + rawPitch * alpha
        
        computeDeltasAndHorizon()
    }
}

/// Free OpenStreetMap (Overpass API) road-type lookup for Vietnam TT38 statutory speed limits.
/// Queries Overpass API to determine if current road is divided/dual-carriageway and fetches maxspeed tags.
@MainActor
final class RoadMetadataService: ObservableObject {
    @Published private(set) var isDividedRoad: Bool = false
    @Published private(set) var roadMaxSpeed: Int? = nil
    @Published private(set) var lastQueryLocation: CLLocation?
    @Published private(set) var isQuerying: Bool = false
    @Published private(set) var lastQueryDate: Date?

    // Throttling: max once per 20s or 150m travel
    private let minQueryInterval: TimeInterval = 20.0
    private let minDistanceMeters: Double = 150.0
    private var lastQueryTime: Date = .distantPast
    private let queryRadius: Double = 25.0  // meters
    private let overpassURL = "https://overpass-api.de/api/interpreter"
    private let requestTimeout: TimeInterval = 8.0

    private let session: URLSession = {
        let config = URLSessionConfiguration.ephemeral
        config.timeoutIntervalForRequest = 8.0
        config.timeoutIntervalForResource = 8.0
        return URLSession(configuration: config)
    }()

    /// Call from LocationManager when location updates significantly.
    func maybeQueryRoadMetadata(for location: CLLocation) {
        let now = Date()
        // Time throttle
        if now.timeIntervalSince(lastQueryTime) < minQueryInterval { return }
        // Distance throttle
        if let lastLoc = lastQueryLocation, location.distance(from: lastLoc) < minDistanceMeters { return }

        lastQueryTime = now
        lastQueryLocation = location
        Task { await queryRoadMetadata(for: location) }
    }

    private func queryRoadMetadata(for location: CLLocation) async {
        isQuerying = true
        defer { isQuerying = false; lastQueryDate = Date() }

        let lat = location.coordinate.latitude
        let lon = location.coordinate.longitude
        let query = """
        [out:json][timeout:5];
        way(around:\(queryRadius),\(lat),\(lon))["highway"];
        out tags 1;
        """

        guard let url = URL(string: overpassURL) else { return }
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.httpBody = query.data(using: .utf8)
        request.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
        request.timeoutInterval = requestTimeout

        do {
            let (data, _) = try await session.data(for: request)
            let result = try parseOverpassResponse(data)
            self.isDividedRoad = result.isDivided
            self.roadMaxSpeed = result.maxSpeed
        } catch {
            // Offline-safe: silently ignore, keep last known values
            print("RoadMetadataService: Overpass query failed: \(error)")
        }
    }

    /// Parse Overpass JSON response for oneway, dual_carriageway, lanes, maxspeed tags.
    private func parseOverpassResponse(_ data: Data) throws -> (isDivided: Bool, maxSpeed: Int?) {
        struct OverpassResponse: Decodable {
            let elements: [Element]
            struct Element: Decodable {
                let tags: [String: String]?
            }
        }
        let response = try JSONDecoder().decode(OverpassResponse.self, from: data)

        var bestHighway: String? = nil
        var bestLanes: Int? = nil
        var bestMaxSpeed: Int? = nil
        var bestOneway: String? = nil
        var bestDualCarriageway: String? = nil

        // Priority: primary > secondary > tertiary > unclassified > residential > service
        let highwayPriority: [String: Int] = [
            "motorway": 7, "trunk": 6, "primary": 5, "secondary": 4,
            "tertiary": 3, "unclassified": 2, "residential": 1, "service": 0
        ]

        for element in response.elements {
            guard let tags = element.tags else { continue }
            let highway = tags["highway"] ?? ""
            let priority = highwayPriority[highway] ?? -1
            let currentBestPriority = highwayPriority[bestHighway ?? ""] ?? -1

            if priority > currentBestPriority {
                bestHighway = highway
                bestLanes = Int(tags["lanes"] ?? "")
                bestMaxSpeed = Int(tags["maxspeed"] ?? "")
                bestOneway = tags["oneway"]
                bestDualCarriageway = tags["dual_carriageway"]
            }
        }

        // Determine divided road status
        var isDivided = false
        if bestOneway == "yes" || bestDualCarriageway == "yes" {
            isDivided = true
        } else if let highway = bestHighway, let lanes = bestLanes, lanes >= 2 {
            // Exclude residential/service unless lanes >= 2 (already checked)
            if highway != "residential" && highway != "service" {
                isDivided = true
            }
        }

        return (isDivided, bestMaxSpeed)
    }

    /// Manual override from driver tap on HUD pill
    func toggleDividedRoad() {
        isDividedRoad.toggle()
    }
}

/// Vietnam traffic sign recognition using custom CoreML model (vn_signs_ios16.mlpackage compiled to .mlmodelc).
/// Runs VNCoreMLRequest at 2 Hz on Neural Engine, maps class labels (Count strings from data.yaml) to Vietnamese sign labels.
final class TrafficSignDetector: ObservableObject {
    // MARK: - Published
    @Published private(set) var detectedSpeedLimit: Int?
    @Published private(set) var isProcessing = false
    @Published private(set) var didClearSpeedLimitEvent: Date?
    @Published private(set) var detectedEndOfAllRestrictions: Bool = false
    @Published private(set) var activeAdvisorySignText: String?  // e.g., "No Overtaking", "No Cars", "Residential Area"
    @Published private(set) var isInResidentialArea: Bool? = nil  // true = R.420 seen, false = R.421/DP.135 seen
    @Published private(set) var isDividedRoadFromSign: Bool = false  // set when class "142" (dual carriageway) detected

    // MARK: - Configuration
    private let processInterval: TimeInterval = 0.5  // 2 Hz
    private let requiredConsecutiveMatches = 2
    private let confidenceThreshold: Float = 0.35
    private let minBoxWidth: CGFloat = 0.018
    // For diagnostic logging: include raw labels at lower confidence to see what model emits
    private let diagnosticConfidenceThreshold: Float = 0.25
    // Tolerate 1 transient missed frame during consecutive confirmation
    private let maxMissedMatchFrames = 3

    // MARK: - Label to Sign mapping (ONLY true Count strings from vn_signs_ios16.mlpackage data.yaml)
    // Speed limits: "124"=ID57->10, "114"=ID61->20, "360"=ID2->40, "402"=ID39->50,
    // "750"=ID12->60, "105"=ID40->70, "441"=ID41->80, "102"=ID63->90, "103"=ID58->100, "101"=ID59->110
    // Residential area (sets 50 km/h + advisory): "164"=ID79 (R.420)
    // End residential area: "117"=ID80 (R.421), "190"=ID48 (DP.135 end all prohibition)
    // Clear speed limit (DP.134): "141"=ID78 (End of 50km/h speed limit)
    // Dual carriageway (sets divided road): "142"=ID73
    // Prohibitory advisory: "572"=ID0->no_entry, "279"=ID6->no_cars, "123"=ID45->no_overtaking
    private let labelToSign: [String: SignInfo] = [
        // Speed limits (Count string -> speed value)
        "124": .speedLimit(10),   // ID 57: Speed limit 10km/h
        "114": .speedLimit(20),   // ID 61: Speed limit 20km/h
        "360": .speedLimit(40),   // ID 2: Speed limit 40km/h
        "402": .speedLimit(50),   // ID 39: Speed limit 50km/h
        "750": .speedLimit(60),   // ID 12: Speed limit 60km/h
        "105": .speedLimit(70),   // ID 40: Speed limit 70km/h
        "441": .speedLimit(80),   // ID 41: Speed limit 80km/h
        "102": .speedLimit(90),   // ID 63: Speed limit 90km/h
        "103": .speedLimit(100),  // ID 58: Speed limit 100km/h
        "101": .speedLimit(110),  // ID 59: Speed limit 110km/h
        // Residential area (sets 50 km/h speed limit + advisory key)
        "164": .residential50,    // ID 79: Residential area (R.420) - ENTER
        // End residential area / end restrictions
        "117": .endResidentialArea,    // ID 80: Sparsely populated area (R.421) - EXIT
        "190": .endResidentialArea,    // ID 48: End of all prohibition (DP.135) - EXIT
        // Clear speed limit (DP.134)
        "141": .clearLimit,       // ID 78: End of 50km/h speed limit
        // Dual carriageway (sets divided road flag)
        "142": .dualCarriageway,  // ID 73: Dual carriageway
        // Prohibitory advisory (canonical keys for localization)
        "572": .advisory("no_entry"),      // ID 0: No Entry
        "279": .advisory("no_cars"),       // ID 6: No Cars
        "123": .advisory("no_overtaking"), // ID 45: No Overtaking
    ]

    private enum SignInfo: Equatable {
        case speedLimit(Int)
        case residential50  // sets detectedSpeedLimit = 50 AND activeAdvisorySignText = "residential_area"
        case endResidentialArea  // R.421 "117" or DP.135 "190" - exit residential area
        case clearLimit
        case dualCarriageway  // "142" - dual carriageway sign
        case advisory(String)  // canonical key: "no_entry", "no_cars", "no_overtaking"

        var isSpeedLimit: Bool { if case .speedLimit = self { return true } else { return false } }
        var isResidential50: Bool { if case .residential50 = self { return true } else { return false } }
        var isEndResidentialArea: Bool { if case .endResidentialArea = self { return true } else { return false } }
        var isClearLimit: Bool { if case .clearLimit = self { return true } else { return false } }
        var isDualCarriageway: Bool { if case .dualCarriageway = self { return true } else { return false } }
        var isAdvisory: Bool { if case .advisory = self { return true } else { return false } }
        var speedValue: Int? { if case .speedLimit(let v) = self { return v } else { return nil } }
        var advisoryKey: String? { if case .advisory(let k) = self { return k } else { return nil } }
    }

    // MARK: - State
    private var lastProcessTime: TimeInterval = 0
    private var consecutiveMatchCount = 0
    private var lastMatchedLimit: Int?
    private var missedMatchFrames = 0  // tolerance for transient missed frame
    private var thermalState: ProcessInfo.ThermalState = .nominal
    private var visionOrientation: CGImagePropertyOrientation = .up
    private var coreMLModel: VNCoreMLModel?
    private var modelLoaded = false
    private var loadTask: Task<Void, Never>?

    // Confirmation state
    private var endSignConsecutiveCount = 0
    private var lastEndSignType: EndSignType?
    private var advisoryConsecutiveCount = 0
    private var lastAdvisoryText: String?
    private var endResidentialConsecutiveCount = 0  // for R.421/DP.135 exit confirmation

    private enum EndSignType { case dp134, dp135 }

    // Diagnostics
    private weak var diagnostics: DiagnosticsLogStore?
    private var lastTSRLogTime: TimeInterval = 0

    init() {
        loadCoreMLModelAsync()

        loadTask?.cancel()

    private func loadCoreMLModelAsync() {
        loadTask = Task.detached(priority: .userInitiated) { [weak self] in
            guard let self else { return }
            do {
                // Xcode compiles .mlpackage resources to .mlmodelc in the app bundle
                let modelURL = Bundle.main.url(forResource: "vn_signs_ios16", withExtension: "mlmodelc")
                guard let modelURL else {
                    print("TrafficSignDetector: vn_signs_ios16.mlmodelc not found in bundle")
                    return
                }
                let mlModel = try MLModel(contentsOf: modelURL, configuration: {
                    let config = MLModelConfiguration()
                    config.computeUnits = .cpuAndNeuralEngine
                    return config
                }())
                let vnModel = try VNCoreMLModel(for: mlModel)
                await MainActor.run {
                    self.coreMLModel = vnModel
                    self.modelLoaded = true
                    print("TrafficSignDetector: CoreML model loaded successfully from \(modelURL.lastPathComponent)")
                }
            } catch {
                print("TrafficSignDetector: Failed to load CoreML model: \(error)")
            }
        }
    }

    /// Attach diagnostics store for TSR event logging
    func attachDiagnostics(_ diagnostics: DiagnosticsLogStore) {
        self.diagnostics = diagnostics
    }

    /// Set vision orientation (matches CameraManager's visionOrientation)
    func setVisionOrientation(_ orientation: CGImagePropertyOrientation) {
        visionOrientation = orientation
    }

    /// Call from CameraManager.captureOutput for each frame.
    func processFrameIfNeeded(_ pixelBuffer: CVPixelBuffer) {
        guard !isProcessing else { return }
        let now = CACurrentMediaTime()
        guard now - lastProcessTime >= processInterval else { return }
        guard thermalState != .critical else { return }
        guard modelLoaded, coreMLModel != nil else { return }

        lastProcessTime = now
        isProcessing = true

        DispatchQueue.global(qos: .userInitiated).async { [weak self] in
            self?.processFrame(pixelBuffer)
        }
    }

    private func processFrame(_ pixelBuffer: CVPixelBuffer) {
        defer { isProcessing = false }

        guard let request = makeCoreMLRequest() else { return }

        let handler = VNImageRequestHandler(cvPixelBuffer: pixelBuffer, orientation: visionOrientation, options: [:])
        do {
            try handler.perform([request])
        } catch {
            print("TrafficSignDetector: VNCoreMLRequest error: \(error)")
            return
        }

        guard let results = request.results as? [VNRecognizedObjectObservation] else { return }

        // Filter detections at normal threshold for state updates
        var speedLimitDetections: [(Int, Float, CGRect)] = []  // (speedValue, confidence, bbox)
        var clearLimitDetections: [(Float, CGRect)] = []        // (confidence, bbox)
        var advisoryDetections: [(String, Float, CGRect)] = []  // (key, confidence, bbox)
        var residential50Detected = false
        var endResidentialAreaDetected = false
        var dualCarriagewayDetected = false

        // For diagnostic logging: collect ALL raw labels at lower threshold
        var allRawDetections: [(String, Float, CGRect)] = []    // (rawLabel, confidence, bbox)

        for obs in results {
            guard obs.confidence >= diagnosticConfidenceThreshold else { continue }
            guard obs.boundingBox.width >= minBoxWidth else { continue }

            // Get the top label - identifier is the class label string (Count from data.yaml)
            guard let topLabel = obs.labels.first else { continue }
            let label = topLabel.identifier
            let bbox = obs.boundingBox  // normalized, bottom-left origin

            // Record ALL raw labels for verification
            allRawDetections.append((label, obs.confidence, bbox))

            // Only process for state updates if above normal confidence threshold
            guard obs.confidence >= confidenceThreshold else { continue }
            guard let signInfo = labelToSign[label] else { continue }

            if signInfo.isSpeedLimit, let speed = signInfo.speedValue {
                speedLimitDetections.append((speed, obs.confidence, bbox))
            } else if signInfo.isResidential50 {
                residential50Detected = true
                speedLimitDetections.append((50, obs.confidence, bbox))
            } else if signInfo.isEndResidentialArea {
                endResidentialAreaDetected = true
            } else if signInfo.isClearLimit {
                clearLimitDetections.append((obs.confidence, bbox))
            } else if signInfo.isDualCarriageway {
                dualCarriagewayDetected = true
            } else if signInfo.isAdvisory, let key = signInfo.advisoryKey {
                advisoryDetections.append((key, obs.confidence, bbox))
            }
        }

        // Diagnostic logging: every 1.0s - include ALL raw labels
        let now = CACurrentMediaTime()
        if now - lastTSRLogTime >= 1.0 {
            lastTSRLogTime = now
            let rawDetails = allRawDetections.map {
                "(rawLabel=\($0.0), conf=\(String(format: "%.2f", $0.1)), x=\(String(format: "%.2f", $0.2.minX)), w=\(String(format: "%.3f", $0.2.width)))"
            }.joined(separator: ";")
            let details = "raw=\(rawDetails.isEmpty ? "none" : rawDetails); speedDetections=\(speedLimitDetections.map { "(\($0.0),\(String(format: "%.2f", $0.1)),\(String(format: "%.3f", $0.2.minX)))\($0.2.width)" }.joined(separator: ";")) clearDetections=\(clearLimitDetections.count) advisoryDetections=\(advisoryDetections.map { "(\($0.0),\(String(format: "%.2f", $0.1)))" }.joined(separator: ";")) residential50=\(residential50Detected) endResidential=\(endResidentialAreaDetected) dualCarriageway=\(dualCarriagewayDetected)"
            diagnostics?.record(type: "tsr_yolo_detect", details: details)
        }

        // Handle speed limit detection (including residential50 which sets 50 km/h)
        if let bestSpeed = speedLimitDetections.max(by: { $0.1 < $1.1 }) {
            let speedValue = bestSpeed.0
            let bestConfidence = bestSpeed.1
            endSignConsecutiveCount = 0
            lastEndSignType = nil
            missedMatchFrames = 0  // reset miss tolerance on detection

            // High-confidence immediate confirmation (>= 0.45) bypasses 2-frame requirement
            let shouldConfirmImmediately = bestConfidence >= 0.45

            if speedValue == lastMatchedLimit {
                consecutiveMatchCount += 1
            } else {
                consecutiveMatchCount = 1
                lastMatchedLimit = speedValue
            }

            if shouldConfirmImmediately || consecutiveMatchCount >= requiredConsecutiveMatches {
                DispatchQueue.main.async { [weak self] in
                    self?.detectedSpeedLimit = speedValue
                    self?.detectedEndOfAllRestrictions = false
                    // If residential50 was detected, also set advisory and residential area flag
                    if residential50Detected {
                        self?.activeAdvisorySignText = "residential_area"
                        self?.isInResidentialArea = true
                        DispatchQueue.main.asyncAfter(deadline: .now() + 3.5) { [weak self] in
                            if self?.activeAdvisorySignText == "residential_area" {
                                self?.activeAdvisorySignText = nil
                            }
                        }
                    }
                }
                diagnostics?.record(type: "tsr_state", details: "action=P127_confirmed limit=\(speedValue) conf=\(String(format: "%.2f", bestConfidence)) immediate=\(shouldConfirmImmediately) consecutive=\(consecutiveMatchCount) residential50=\(residential50Detected)")
            }
        } else {
            // Tolerate transient missed frames before resetting confirmation
            if lastMatchedLimit != nil {
                missedMatchFrames += 1
                if missedMatchFrames > maxMissedMatchFrames {
                    consecutiveMatchCount = 0
                    lastMatchedLimit = nil
                    missedMatchFrames = 0
                }
            }
        }

        // Handle clear limit detection (DP.134 / DP.135)
        if !clearLimitDetections.isEmpty && detectedSpeedLimit != nil {
            handleEndSignDetection(.dp134)
        } else {
            endSignConsecutiveCount = 0
            lastEndSignType = nil
        }

        // Handle advisory sign detection
        if let bestAdvisory = advisoryDetections.max(by: { $0.1 < $1.1 }) {
            let advisoryKey = bestAdvisory.0
            let advisoryConfidence = bestAdvisory.1
            if advisoryKey == lastAdvisoryText {
                advisoryConsecutiveCount += 1
            } else {
                advisoryConsecutiveCount = 1
                lastAdvisoryText = advisoryKey
            }

            // High-confidence immediate confirmation (>= 0.45) bypasses 2-frame requirement
            let shouldConfirmImmediately = advisoryConfidence >= 0.45

            if shouldConfirmImmediately || advisoryConsecutiveCount >= requiredConsecutiveMatches {
                DispatchQueue.main.async { [weak self] in
                    self?.activeAdvisorySignText = advisoryKey
                    // Auto-clear after 5.0s for transient prohibitory signs
                    DispatchQueue.main.asyncAfter(deadline: .now() + 5.0) { [weak self] in
                        if self?.activeAdvisorySignText == advisoryKey {
                            self?.activeAdvisorySignText = nil
                        }
                    }
                }
                diagnostics?.record(type: "tsr_state", details: "action=advisory_confirmed key=\(advisoryKey) conf=\(String(format: "%.2f", advisoryConfidence)) immediate=\(shouldConfirmImmediately)")
            }
        } else {
            advisoryConsecutiveCount = 0
            lastAdvisoryText = nil
        }

        // Handle end of residential area detection (R.421 "117" or DP.135 "190") — require 2 consecutive matches
        if endResidentialAreaDetected {
            endResidentialConsecutiveCount += 1
            if endResidentialConsecutiveCount >= requiredConsecutiveMatches {
                DispatchQueue.main.async { [weak self] in
                    self?.detectedSpeedLimit = nil
                    self?.didClearSpeedLimitEvent = Date()
                    self?.isInResidentialArea = false
                }
                diagnostics?.record(type: "tsr_state", details: "action=end_residential_area_confirmed cleared_limit")
                endResidentialConsecutiveCount = 0
            }
        } else {
            endResidentialConsecutiveCount = 0
        }

        // Handle dual carriageway detection (class "142")
        if dualCarriagewayDetected {
            DispatchQueue.main.async { [weak self] in
                self?.isDividedRoadFromSign = true
            }
            diagnostics?.record(type: "tsr_state", details: "action=dual_carriageway confirmed")
        }
    }

import CoreLocation
import Foundation

final class WeatherService {
    private let queue = DispatchQueue(label: "drive-assistant.weather")
    private var lastRequestDate: Date?
    private var lastRequestLocation: CLLocation?
    private let session: URLSession

    init() {
        let config = URLSessionConfiguration.ephemeral
        config.timeoutIntervalForRequest = 8
        config.timeoutIntervalForResource = 10
        config.waitsForConnectivity = false
        self.session = URLSession(configuration: config)
    }

    func update(for location: CLLocation, completion: @escaping (RoadCondition) -> Void) {
        queue.async {
            if let lastRequestDate = self.lastRequestDate,
               let lastRequestLocation = self.lastRequestLocation,
               Date().timeIntervalSince(lastRequestDate) < 300,
               location.distance(from: lastRequestLocation) <= 5_000 {
                return
            }

            self.lastRequestDate = Date()
            self.lastRequestLocation = location
            guard var components = URLComponents(string: "https://api.open-meteo.com/v1/forecast") else {
                DispatchQueue.main.async { completion(.dry) }
                return
            }
            components.queryItems = [
                URLQueryItem(name: "latitude", value: String(location.coordinate.latitude)),
                URLQueryItem(name: "longitude", value: String(location.coordinate.longitude)),
                URLQueryItem(name: "current", value: "precipitation,rain,weather_code")
            ]
            guard let url = components.url else {
                DispatchQueue.main.async { completion(.dry) }
                return
            }

            self.session.dataTask(with: url) { data, response, _ in
                guard let httpResponse = response as? HTTPURLResponse,
                      (200...299).contains(httpResponse.statusCode),
                      let data,
                      let payload = try? JSONDecoder().decode(OpenMeteoResponse.self, from: data) else {
                    DispatchQueue.main.async { completion(.dry) }
                    return
                }
                let current = payload.current
                let isWet = current.precipitation > 0.1 || current.rain > 0.1 || current.weatherCode >= 51
                DispatchQueue.main.async { completion(isWet ? .wet : .dry) }
            }.resume()
        }
    }
}

private struct OpenMeteoResponse: Decodable {
    let current: CurrentWeather

    struct CurrentWeather: Decodable {
        let precipitation: Double
        let rain: Double
        let weatherCode: Int

        enum CodingKeys: String, CodingKey {
            case precipitation
            case rain
            case weatherCode = "weather_code"
        }
    }
}
